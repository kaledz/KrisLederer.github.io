Data Transformation Unit Tests

# tests/unit/test_transformations.py
import pytest
from pyspark.sql.functions import col, when
from pyspark.sql.types import StructType, StructField, StringType, IntegerType

# Import your transformation functions
from src.data_pipeline.transformations import (
    clean_employee_data,
    calculate_salary_bands,
    aggregate_department_stats,
    validate_data_quality
)

class TestDataTransformations:
    
    def test_clean_employee_data(self, spark, sample_df):
        """Test data cleaning function"""
        
        # Add some dirty data to test cleaning
        dirty_data = [
            {"id": 6, "name": " JOHN ", "age": None, "department": "", "salary": -1000},
            {"id": 7, "name": "jane doe", "age": 29, "department": None, "salary": 55000}
        ]
        
        dirty_df = spark.createDataFrame(dirty_data)
        combined_df = sample_df.union(dirty_df)
        
        # Apply cleaning function
        cleaned_df = clean_employee_data(combined_df)
        
        # Assertions
        assert cleaned_df.count() > 0, "Cleaned dataframe should not be empty"
        
        # Check that names are properly formatted
        names = [row.name for row in cleaned_df.select("name").collect()]
        assert all(name.strip() == name for name in names), "Names should be stripped"
        assert all(name.title() == name for name in names), "Names should be title case"
        
        # Check that null ages are handled
        null_ages = cleaned_df.filter(col("age").isNull()).count()
        assert null_ages == 0, "Should handle null ages"
        
        # Check that negative salaries are handled
        negative_salaries = cleaned_df.filter(col("salary") < 0).count()
        assert negative_salaries == 0, "Should handle negative salaries"
    
    def test_calculate_salary_bands(self, spark, sample_df):
        """Test salary band calculation"""
        
        result_df = calculate_salary_bands(sample_df)
        
        # Check that salary_band column is added
        assert "salary_band" in result_df.columns
        
        # Collect results for detailed testing
        results = result_df.collect()
        
        # Test specific salary band logic
        for row in results:
            if row.salary < 60000:
                assert row.salary_band == "Junior"
            elif row.salary < 75000:
                assert row.salary_band == "Mid"
            else:
                assert row.salary_band == "Senior"
    
    def test_aggregate_department_stats(self, spark, sample_df):
        """Test department aggregation"""
        
        agg_df = aggregate_department_stats(sample_df)
        
        # Check expected columns
        expected_columns = ["department", "employee_count", "avg_salary", "total_salary"]
        for col_name in expected_columns:
            assert col_name in agg_df.columns
        
        # Test specific aggregations
        engineering_stats = agg_df.filter(col("department") == "Engineering").collect()[0]
        
        assert engineering_stats.employee_count == 3  # Alice, Charlie, Eve
        assert engineering_stats.avg_salary == 80000  # (75000 + 85000 + 80000) / 3
        assert engineering_stats.total_salary == 240000
    
    def test_validate_data_quality_success(self, spark, sample_df):
        """Test data quality validation with good data"""
        
        quality_result = validate_data_quality(sample_df)
        
        assert quality_result["is_valid"] == True
        assert quality_result["null_percentage"] < 0.1  # Less than 10%
        assert quality_result["duplicate_count"] == 0
        assert len(quality_result["issues"]) == 0
    
    def test_validate_data_quality_failure(self, spark):
        """Test data quality validation with bad data"""
        
        bad_data = [
            {"id": 1, "name": None, "age": None, "department": "Eng", "salary": 75000},
            {"id": 1, "name": None, "age": 25, "department": "Eng", "salary": 75000},  # Duplicate
            {"id": 2, "name": "Bob", "age": -5, "department": None, "salary": None}
        ]
        
        bad_df = spark.createDataFrame(bad_data)
        quality_result = validate_data_quality(bad_df)
        
        assert quality_result["is_valid"] == False
        assert quality_result["null_percentage"] > 0.3  # More than 30%
        assert quality_result["duplicate_count"] > 0
        assert len(quality_result["issues"]) > 0
    
    def test_error_handling(self, spark):
        """Test error handling with invalid input"""
        
        with pytest.raises(Exception) as exc_info:
            # Test with DataFrame missing required columns
            incomplete_df = spark.createDataFrame([{"id": 1, "name": "Test"}])
            clean_employee_data(incomplete_df)
        
        assert "required column" in str(exc_info.value).lower()

    def test_schema_validation(self, spark):
        """Test that transformations preserve expected schema"""
        
        # Define expected schema
        expected_schema = StructType([
            StructField("id", IntegerType(), True),
            StructField("name", StringType(), True),
            StructField("age", IntegerType(), True),
            StructField("department", StringType(), True),
            StructField("salary", IntegerType(), True),
            StructField("salary_band", StringType(), True)
        ])
        
        sample_data = [{"id": 1, "name": "Test", "age": 30, "department": "IT", "salary": 70000}]
        test_df = spark.createDataFrame(sample_data)
        
        result_df = calculate_salary_bands(clean_employee_data(test_df))
        
        # Check schema compatibility
        assert len(result_df.schema) == len(expected_schema)
        
        for expected_field, actual_field in zip(expected_schema, result_df.schema):
            assert expected_field.name == actual_field.name
            assert expected_field.dataType == actual_field.dataType
  • Comprehensive testing: Tests both happy path and edge cases
  • Data quality validation: Ensures transformations handle dirty data correctly
  • Schema validation: Verifies output schema matches expectations
  • Error handling: Tests that functions fail gracefully with invalid input

# tests/integration/test_pipeline.py
import pytest
import tempfile
import shutil
from pathlib import Path

from src.data_pipeline.pipeline import ETLPipeline
from src.data_pipeline.config import load_config

class TestETLPipeline:
    
    @pytest.fixture
    def temp_data_dir(self):
        """Create temporary directory for test data"""
        temp_dir = tempfile.mkdtemp()
        yield temp_dir
        shutil.rmtree(temp_dir)
    
    @pytest.fixture
    def test_config(self, temp_data_dir):
        """Create test configuration"""
        return {
            "input_path": f"{temp_data_dir}/input",
            "output_path": f"{temp_data_dir}/output",
            "checkpoint_path": f"{temp_data_dir}/checkpoints",
            "batch_size": 100,
            "parallelism": 2
        }
    
    def test_full_pipeline_execution(self, spark, sample_data, temp_data_dir, test_config):
        """Test complete pipeline execution"""
        
        # Setup: Create input data
        input_df = spark.createDataFrame(sample_data)
        input_df.coalesce(1).write.mode("overwrite").parquet(test_config["input_path"])
        
        # Execute pipeline
        pipeline = ETLPipeline(spark, test_config)
        result = pipeline.run()
        
        # Verify results
        assert result["status"] == "SUCCESS"
        assert result["records_processed"] == len(sample_data)
        
        # Verify output data exists and is correct
        output_df = spark.read.parquet(test_config["output_path"])
        
        assert output_df.count() == len(sample_data)
        assert "salary_band" in output_df.columns  # Transformation applied
        assert output_df.filter(col("salary_band").isNull()).count() == 0
    
    def test_pipeline_with_invalid_input(self, spark, test_config):
        """Test pipeline behavior with invalid input"""
        
        # No input data scenario
        pipeline = ETLPipeline(spark, test_config)
        
        with pytest.raises(Exception) as exc_info:
            pipeline.run()
        
        assert "input path not found" in str(exc_info.value).lower()
    
    def test_pipeline_data_quality_failure(self, spark, temp_data_dir, test_config):
        """Test pipeline with data quality issues"""
        
        # Create problematic input data
        bad_data = [
            {"id": None, "name": None, "age": None, "department": None, "salary": None}
        ] * 100  # 100 completely null records
        
        bad_df = spark.createDataFrame(bad_data)
        bad_df.coalesce(1).write.mode("overwrite").parquet(test_config["input_path"])
        
        # Configure pipeline to fail on quality issues
        test_config["fail_on_quality_issues"] = True
        test_config["max_null_percentage"] = 0.1  # 10% threshold
        
        pipeline = ETLPipeline(spark, test_config)
        
        with pytest.raises(Exception) as exc_info:
            pipeline.run()
        
        assert "data quality" in str(exc_info.value).lower()
    
    def test_pipeline_performance(self, spark, temp_data_dir, test_config):
        """Test pipeline performance with larger dataset"""
        
        # Create larger test dataset
        large_data = []
        for i in range(10000):
            large_data.append({
                "id": i,
                "name": f"Employee_{i}",
                "age": 25 + (i % 40),
                "department": ["Engineering", "Sales", "Marketing"][i % 3],
                "salary": 50000 + (i % 50000)
            })
        
        large_df = spark.createDataFrame(large_data)
        large_df.coalesce(4).write.mode("overwrite").parquet(test_config["input_path"])
        
        # Execute pipeline with timing
        import time
        start_time = time.time()
        
        pipeline = ETLPipeline(spark, test_config)
        result = pipeline.run()
        
        execution_time = time.time() - start_time
        
        # Performance assertions
        assert result["status"] == "SUCCESS"
        assert result["records_processed"] == 10000
        assert execution_time < 120  # Should complete in under 2 minutes
        
        # Verify output
        output_df = spark.read.parquet(test_config["output_path"])
        assert output_df.count() == 10000
        
    def test_pipeline_idempotency(self, spark, sample_data, temp_data_dir, test_config):
        """Test that pipeline can be run multiple times safely"""
        
        # Setup input data
        input_df = spark.createDataFrame(sample_data)
        input_df.coalesce(1).write.mode("overwrite").parquet(test_config["input_path"])
        
        pipeline = ETLPipeline(spark, test_config)
        
        # Run pipeline first time
        result1 = pipeline.run()
        output_count_1 = spark.read.parquet(test_config["output_path"]).count()
        
        # Run pipeline second time (should overwrite)
        result2 = pipeline.run()
        output_count_2 = spark.read.parquet(test_config["output_path"]).count()
        
        # Both runs should succeed and produce same results
        assert result1["status"] == "SUCCESS"
        assert result2["status"] == "SUCCESS"
        assert output_count_1 == output_count_2 == len(sample_data)

  • Performance testing: Validates pipeline performance with realistic data volumes
  • Error scenarios: Tests how pipeline handles various failure conditions

Continuous Integration Workflow

# .github/workflows/ci-test.yml
name: Continuous Integration

on:
  push:
    branches: [ main, develop ]
  pull_request:
    branches: [ main ]

env:
  PYTHON_VERSION: '3.9'
  JAVA_VERSION: '11'

jobs:
  lint-and-format:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: ${{ env.PYTHON_VERSION }}
    
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install black flake8 mypy isort bandit
        pip install -r requirements-dev.txt
    
    - name: Format check with Black
      run: black --check --diff src/ tests/
    
    - name: Import sorting check
      run: isort --check-only --diff src/ tests/
    
    - name: Lint with flake8
      run: |
        flake8 src/ tests/ --count --select=E9,F63,F7,F82 --show-source --statistics
        flake8 src/ tests/ --count --max-complexity=10 --max-line-length=88 --statistics
    
    - name: Type checking with mypy
      run: mypy src/
    
    - name: Security scan with bandit
      run: bandit -r src/ -f json -o bandit-report.json
    
    - name: Upload security scan results
      uses: actions/upload-artifact@v3
      if: always()
      with:
        name: security-scan-results
        path: bandit-report.json

  unit-tests:
    runs-on: ubuntu-latest
    needs: lint-and-format
    
    strategy:
      matrix:
        python-version: ['3.8', '3.9', '3.10']
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python ${{ matrix.python-version }}
      uses: actions/setup-python@v4
      with:
        python-version: ${{ matrix.python-version }}
    
    - name: Set up Java
      uses: actions/setup-java@v3
      with:
        distribution: 'temurin'
        java-version: ${{ env.JAVA_VERSION }}
    
    - name: Cache pip dependencies
      uses: actions/cache@v3
      with:
        path: ~/.cache/pip
        key: ${{ runner.os }}-pip-${{ hashFiles('**/requirements*.txt') }}
        restore-keys: |
          ${{ runner.os }}-pip-
    
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt
        pip install -r requirements-dev.txt
        pip install -e .
    
    - name: Run unit tests
      run: |
        pytest tests/unit/ -v \
          --cov=src/ \
          --cov-report=xml \
          --cov-report=html \
          --junitxml=pytest-results.xml
    
    - name: Upload test results
      uses: actions/upload-artifact@v3
      if: always()
      with:
        name: pytest-results-${{ matrix.python-version }}
        path: |
          pytest-results.xml
          htmlcov/
    
    - name: Upload coverage to Codecov
      uses: codecov/codecov-action@v3
      with:
        file: coverage.xml
        flags: unittests
        name: codecov-${{ matrix.python-version }}

  integration-tests:
    runs-on: ubuntu-latest
    needs: unit-tests
    
    services:
      # Add any external services needed for integration tests
      postgres:
        image: postgres:13
        env:
          POSTGRES_PASSWORD: testpassword
          POSTGRES_DB: testdb
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: ${{ env.PYTHON_VERSION }}
    
    - name: Set up Java
      uses: actions/setup-java@v3
      with:
        distribution: 'temurin'
        java-version: ${{ env.JAVA_VERSION }}
    
    - name: Install dependencies
      run: |
        python -m pip install --upgrade pip
        pip install -r requirements.txt
        pip install -r requirements-dev.txt
        pip install -e .
    
    - name: Run integration tests
      env:
        DATABASE_URL: postgresql://postgres:testpassword@localhost:5432/testdb
      run: |
        pytest tests/integration/ -v \
          --cov=src/ \
          --cov-append \
          --cov-report=xml
    
    - name: Upload integration test results
      uses: actions/upload-artifact@v3
      if: always()
      with:
        name: integration-test-results
        path: coverage.xml

  build-package:
    runs-on: ubuntu-latest
    needs: [unit-tests, integration-tests]
    if: github.ref == 'refs/heads/main'
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: ${{ env.PYTHON_VERSION }}
    
    - name: Install build tools
      run: |
        python -m pip install --upgrade pip
        pip install build twine
    
    - name: Build package
      run: python -m build
    
    - name: Check package
      run: twine check dist/*
    
    - name: Upload build artifacts
      uses: actions/upload-artifact@v3
      with:
        name: python-package
        path: dist/
        retention-days: 30

  notify-status:
    runs-on: ubuntu-latest
    needs: [lint-and-format, unit-tests, integration-tests, build-package]
    if: always()
    
    steps:
    - name: Notify success
      if: ${{ needs.lint-and-format.result == 'success' && needs.unit-tests.result == 'success' && needs.integration-tests.result == 'success' }}
      uses: 8398a7/action-slack@v3
      with:
        status: success
        text: "✅ CI Pipeline completed successfully for ${{ github.ref }}"
      env:

Continuous Deployment - Staging

# .github/workflows/cd-staging.yml
name: Deploy to Staging

on:
  push:
    branches: [ develop ]
  workflow_run:
    workflows: ["Continuous Integration"]
    branches: [ develop ]
    types: [ completed ]

env:
  DATABRICKS_HOST: ${{ secrets.DATABRICKS_STAGING_HOST }}
  DATABRICKS_TOKEN: ${{ secrets.DATABRICKS_STAGING_TOKEN }}

jobs:
  deploy-staging:
    runs-on: ubuntu-latest
    if: ${{ github.event.workflow_run.conclusion == 'success' }}
    
    environment:
      name: staging
      url: ${{ steps.deploy.outputs.job-url }}
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Set up Python
      uses: actions/setup-python@v4
      with:
        python-version: '3.9'
    
    - name: Install Databricks CLI
      run: |
        pip install databricks-cli
        pip install -r requirements.txt
    
    - name: Configure Databricks CLI
      run: |
        echo "[staging]" > ~/.databrickscfg
        echo "host = $DATABRICKS_HOST" >> ~/.databrickscfg
        echo "token = $DATABRICKS_TOKEN" >> ~/.databrickscfg
    
    - name: Download build artifacts
      if: github.event_name == 'workflow_run'
      uses: actions/download-artifact@v3
      with:
        name: python-package
        path: dist/
        run-id: ${{ github.event.workflow_run.id }}
    
    - name: Build package (if direct push)
      if: github.event_name == 'push'
      run: |
        pip install build
        python -m build
    
    - name: Upload notebooks to Databricks
      run: |
        databricks workspace import-dir notebooks /Workspace/Repos/staging/data-pipeline/notebooks \
          --profile staging \
          --overwrite \
          --language PYTHON
    
    - name: Upload package to DBFS
      run: |
        databricks fs cp dist/*.whl dbfs:/shared/packages/staging/ \
          --profile staging \
          --overwrite
    
    - name: Update job configuration
      id: deploy
      run: |
        # Replace placeholders in job config with staging values
        envsubst < jobs/daily_etl_job.json > jobs/staging_daily_etl_job.json
        
        # Update or create the job
        job_id=$(databricks jobs list --profile staging --output JSON | \
          jq -r '.jobs[] | select(.settings.name=="Staging Daily ETL") | .job_id')
        
        if [ "$job_id" != "null" ] && [ -n "$job_id" ]; then
          echo "Updating existing job $job_id"
          databricks jobs reset --profile staging --job-id $job_id --json-file jobs/staging_daily_etl_job.json
        else
          echo "Creating new job"
          job_id=$(databricks jobs create --profile staging --json-file jobs/staging_daily_etl_job.json | jq -r '.job_id')
        fi
        
        echo "job-url=https://$DATABRICKS_HOST/#job/$job_id" >> $GITHUB_OUTPUT
    
    - name: Run smoke tests
      run: |
        # Run a simple test job to verify deployment
        job_id=$(databricks jobs list --profile staging --output JSON | \
          jq -r '.jobs[] | select(.settings.name=="Staging Daily ETL") | .job_id')
        
        run_id=$(databricks jobs run-now --profile staging --job-id $job_id | jq -r '.run_id')
        
        echo "Started test run: $run_id"
        
        # Wait for completion (with timeout)
        timeout=600  # 10 minutes
        elapsed=0
        
        while [ $elapsed -lt $timeout ]; do
          status=$(databricks runs get --profile staging --run-id $run_id | jq -r '.state.life_cycle_state')
          
          if [ "$status" = "TERMINATED" ]; then
            result_state=$(databricks runs get --profile staging --run-id $run_id | jq -r '.state.result_state')
            
            if [ "$result_state" = "SUCCESS" ]; then
              echo "✅ Smoke test passed"
              break
            else
              echo "❌ Smoke test failed"
              exit 1
            fi
          fi
          
          sleep 30
          elapsed=$((elapsed + 30))
        done
        
        if [ $elapsed -ge $timeout ]; then
          echo "⏰ Smoke test timed out"
          exit 1
        fi
    
    - name: Notify deployment success
      if: success()
      uses: 8398a7/action-slack@v3
      with:
        status: success
        text: "🚀 Successfully deployed to Staging environment"
      env:
        SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK }}
    
    - name: Notify deployment failure
      if: failure()
      uses: 8398a7/action-slack@v3
      with:
        status: failure
        text: "💥 Staging deployment failed"
      env:
        SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK }}
