🔐 Authentication & Setup

Authentication methods for Databricks REST API

  • Personal Access Tokens (recommended for development)
  • OAuth 2.0 (recommended for production)
  • Service Principal authentication

Personal Access Token Setup

# Set environment variables
export DATABRICKS_HOST="https://your-workspace.cloud.databricks.com"
export DATABRICKS_TOKEN="dapi1234567890abcdef"

# Or use databricks CLI configuration
databricks configure --token

Python API Client Setup

# Install Databricks SDK
# pip install databricks-sdk

from databricks.sdk import WorkspaceClient
import requests
import json

# Initialize client with token
w = WorkspaceClient(
    host="https://your-workspace.cloud.databricks.com",
    token="dapi1234567890abcdef"
)

# Alternative: Direct REST API calls
headers = {
    'Authorization': f'Bearer {token}',
    'Content-Type': 'application/json'
}

def make_api_call(endpoint, method='GET', data=None):
    """Generic API call function"""
    url = f"{DATABRICKS_HOST}/api/2.1/{endpoint}"
    
    if method == 'GET':
        response = requests.get(url, headers=headers)
    elif method == 'POST':
        response = requests.post(url, headers=headers, json=data)
    elif method == 'DELETE':
        response = requests.delete(url, headers=headers)
    
    return response.json() if response.status_code == 200 else response.text

Create a New Cluster

POST /api/2.1/clusters/create 
# Create cluster configuration
cluster_config = {
    "cluster_name": "my-analytics-cluster",
    "spark_version": "13.3.x-scala2.12",
    "node_type_id": "i3.xlarge",
    "driver_node_type_id": "i3.xlarge",
    "num_workers": 2,
    "autotermination_minutes": 60,
    "enable_elastic_disk": True,
    "data_security_mode": "SINGLE_USER",
    "runtime_engine": "STANDARD",
    "spark_conf": {
        "spark.sql.adaptive.enabled": "true",
        "spark.sql.adaptive.coalescePartitions.enabled": "true"
    },
    "custom_tags": {
        "environment": "development",
        "team": "data-engineering"
    }
}

# Using SDK
cluster = w.clusters.create(**cluster_config)
print(f"Created cluster: {cluster.cluster_id}")

# Using direct API
response = make_api_call('clusters/create', 'POST', cluster_config)
print(f"Cluster ID: {response.get('cluster_id')}")

# Start cluster
def start_cluster(cluster_id):
    """Start a stopped cluster"""
    w.clusters.start(cluster_id)
    print(f"Starting cluster: {cluster_id}")

# Stop cluster  
def stop_cluster(cluster_id):
    """Stop a running cluster"""
    w.clusters.terminate(cluster_id)
    print(f"Stopping cluster: {cluster_id}")

# Delete cluster
def delete_cluster(cluster_id):
    """Permanently delete a cluster"""
    w.clusters.permanent_delete(cluster_id)
    print(f"Deleted cluster: {cluster_id}")

# Batch operations
def manage_clusters_by_tag(tag_key, tag_value, action):
    """Start/stop/delete clusters by tag"""
    clusters = w.clusters.list()
    for cluster in clusters:
        if cluster.custom_tags and cluster.custom_tags.get(tag_key) == tag_value:
            if action == 'start':
                start_cluster(cluster.cluster_id)
            elif action == 'stop':
                stop_cluster(cluster.cluster_id)
            elif action == 'delete':
                delete_cluster(cluster.cluster_id)

warehouse_config = {

# Start SQL warehouse
def start_sql_warehouse(warehouse_id):
    """Start a stopped SQL warehouse"""
    w.warehouses.start(warehouse_id)
    print(f"Starting warehouse: {warehouse_id}")

# Stop SQL warehouse
def stop_sql_warehouse(warehouse_id):
    """Stop a running SQL warehouse"""
    w.warehouses.stop(warehouse_id)
    print(f"Stopping warehouse: {warehouse_id}")

# Update warehouse configuration
def update_warehouse(warehouse_id, **updates):
    """Update warehouse settings"""
    w.warehouses.edit(warehouse_id, **updates)
    print(f"Updated warehouse: {warehouse_id}")

# Batch warehouse management
def manage_warehouses_by_tag(tag_key, tag_value, action):
    """Start/stop warehouses based on tags"""
    warehouses = w.warehouses.list()
    
    for warehouse in warehouses:
        if warehouse.tags and warehouse.tags.get(tag_key) == tag_value:
            if action == 'start':
                start_sql_warehouse(warehouse.id)
            elif action == 'stop':
                stop_sql_warehouse(warehouse.id)
            print(f"{action.capitalize()}ed warehouse: {warehouse.name}")

# Example: Stop all development warehouses at end of day
manage_warehouses_by_tag('environment', 'development', 'stop')

🏊 Cluster Pools

Manage cluster pools for faster cluster startup times and cost optimization

  • Create and configure cluster pools
  • Monitor pool utilization and costs
  • Scale pools up/down based on demand
  • Delete unused pools to save costs

Create and Manage Cluster Pools

POST /api/2.1/instance-pools/create 
# Create cluster pool
pool_config = {
    "instance_pool_name": "general-purpose-pool",
    "min_idle_instances": 0,
    "max_capacity": 10,
    "node_type_id": "i3.xlarge",
    "idle_instance_autotermination_minutes": 60,
    "enable_elastic_disk": True,
    "custom_tags": {
        "pool-type": "general-purpose",
        "team": "data-engineering"
    }
}

pool = w.instance_pools.create(**pool_config)
print(f"Created pool: {pool.instance_pool_id}")

# List all pools
def list_instance_pools():
    """List all instance pools with stats"""
    pools = w.instance_pools.list()
    for pool in pools:
        stats = pool.stats
        print(f"Pool: {pool.instance_pool_name}")
        print(f"  State: {pool.state}")
        print(f"  Idle: {stats.idle_count}, Used: {stats.used_count}")
        print(f"  Max Capacity: {pool.max_capacity}")
        
        # Calculate utilization
        utilization = (stats.used_count / pool.max_capacity * 100) if pool.max_capacity > 0 else 0
        print(f"  Utilization: {utilization:.1f}%")
        print("-" * 40)

# Update pool configuration
def update_instance_pool(pool_id, **updates):
    """Update pool settings like capacity or idle termination"""
    w.instance_pools.edit(pool_id, **updates)
    print(f"Updated pool: {pool_id}")

# Delete pool
def delete_instance_pool(pool_id):
    """Delete an instance pool"""
    w.instance_pools.delete(pool_id)
    print(f"Deleted pool: {pool_id}")

# Cost management: Scale down pools after hours
def scale_down_pools():
    """Scale down pools for cost optimization"""
    pools = w.instance_pools.list()
    
    for pool in pools:
        # Update to minimum capacity
        update_instance_pool(
            pool.instance_pool_id,
            min_idle_instances=0,
            max_capacity=max(1, pool.max_capacity // 2)  # Reduce capacity by half
        )
        print(f"Scaled down pool: {pool.instance_pool_name}")

# Example usage
list_instance_pools()
scale_down_pools()

💰 Cost Optimization Automation

Automated cost control through API-driven resource management

  • Scheduled shutdown of development resources
  • Weekend cost reduction automation
  • Automatic scaling based on utilization
  • Cost monitoring and alerting

Automated Cost Control Scripts

# Complete cost optimization automation
import schedule
import time
from datetime import datetime, time as dt_time

class DatabricksCostController:
    def __init__(self, workspace_client):
        self.w = workspace_client
    
    def shutdown_dev_resources(self):
        """Shutdown all development resources at end of day"""
        print(f"[{datetime.now()}] Starting dev resource shutdown...")
        
        # Stop development clusters
        clusters = self.w.clusters.list()
        for cluster in clusters:
            if cluster.custom_tags and cluster.custom_tags.get('environment') == 'development':
                if cluster.state in ['RUNNING', 'PENDING']:
                    self.w.clusters.terminate(cluster.cluster_id)
                    print(f"Stopped dev cluster: {cluster.cluster_name}")
        
        # Stop development warehouses
        warehouses = self.w.warehouses.list()
        for warehouse in warehouses:
            if warehouse.tags and warehouse.tags.get('environment') == 'development':
                if warehouse.state == 'RUNNING':
                    self.w.warehouses.stop(warehouse.id)
                    print(f"Stopped dev warehouse: {warehouse.name}")
        
        print("Dev resource shutdown complete!")
    
    def weekend_scale_down(self):
        """Scale down resources for weekend cost savings"""
        print(f"[{datetime.now()}] Starting weekend scale-down...")
        
        # Scale down production warehouses
        warehouses = self.w.warehouses.list()
        for warehouse in warehouses:
            if warehouse.tags and warehouse.tags.get('environment') == 'production':
                # Reduce to minimum clusters
                self.w.warehouses.edit(
                    warehouse.id,
                    min_num_clusters=1,
                    max_num_clusters=2  # Reduce from normal capacity
                )
                print(f"Scaled down warehouse: {warehouse.name}")
        
        # Scale down instance pools
        pools = self.w.instance_pools.list()
        for pool in pools:
            if pool.max_capacity > 2:
                self.w.instance_pools.edit(
                    pool.instance_pool_id,
                    min_idle_instances=0,
                    max_capacity=max(2, pool.max_capacity // 3)
                )
                print(f"Scaled down pool: {pool.instance_pool_name}")
    
    def monday_scale_up(self):
        """Scale resources back up for the work week"""
        print(f"[{datetime.now()}] Starting Monday scale-up...")
        
        # Restore production warehouse capacity
        warehouses = self.w.warehouses.list()
        for warehouse in warehouses:
            if warehouse.tags and warehouse.tags.get('environment') == 'production':
                self.w.warehouses.edit(
                    warehouse.id,
                    min_num_clusters=1,
                    max_num_clusters=5  # Restore full capacity
                )
                print(f"Scaled up warehouse: {warehouse.name}")
    
    def setup_schedule(self):
        """Setup automated cost control schedule"""
        # Daily shutdown of dev resources at 7 PM
        schedule.every().day.at("19:00").do(self.shutdown_dev_resources)
        
        # Weekend scale-down Friday at 6 PM
        schedule.every().friday.at("18:00").do(self.weekend_scale_down)
        
        # Monday scale-up at 7 AM
        schedule.every().monday.at("07:00").do(self.monday_scale_up)
        
        print("Cost optimization schedule configured!")
        print("- Dev resources shutdown: Daily 7 PM")
        print("- Weekend scale-down: Friday 6 PM")  
        print("- Monday scale-up: Monday 7 AM")
    
    def run_scheduler(self):
        """Run the cost optimization scheduler"""
        self.setup_schedule()
        
        while True:
            schedule.run_pending()
            time.sleep(60)  # Check every minute

# Usage
cost_controller = DatabricksCostController(w)

# Run immediate cost optimization
cost_controller.shutdown_dev_resources()

# cost_controller.run_scheduler()

📦 Asset Bundles

Deploy and manage Databricks Asset Bundles (DABs)

  • Package notebooks, jobs, and configurations
  • Deploy to different environments (dev/staging/prod)
  • Version control for complete workspace assets
  • Automated CI/CD with asset bundles

Asset Bundle Configuration

# databricks.yml - Asset Bundle configuration
bundle:
  name: data-pipeline
  compute_id: ${var.cluster_id}
  
variables:
  cluster_id:
    description: "Compute cluster ID"

resources:
  jobs:
    daily_etl:
      name: "Daily ETL Pipeline - ${bundle.environment}"
      schedule:
        quartz_cron_expression: "0 0 2 * * ?"
        timezone_id: "UTC"
      tasks:
        - task_key: extract
          notebook_task:
            notebook_path: ./notebooks/extract.py
          new_cluster:
            spark_version: "13.3.x-scala2.12"
            node_type_id: "i3.xlarge"
            num_workers: 2

targets:
  development:
    workspace:
      host: https://dev-workspace.cloud.databricks.com
    variables:
      cluster_id: "dev-cluster-id"
  
  production:
    workspace:
      host: https://prod-workspace.cloud.databricks.com
    variables:
      cluster_id: "prod-cluster-id"

Deploy Asset Bundle

# Install Databricks CLI with bundle support
pip install databricks-cli

# Initialize new bundle
databricks bundle init

# Validate bundle configuration
databricks bundle validate -t development

# Deploy to development environment
databricks bundle deploy -t development

# Deploy to production environment
databricks bundle deploy -t production

# Run deployed job
databricks bundle run daily_etl -t development

🐍 Python Quick Scripts

# Quick cluster start/stop script
import requests
import os

DATABRICKS_HOST = os.getenv('DATABRICKS_HOST')
TOKEN = os.getenv('DATABRICKS_TOKEN')
headers = {'Authorization': f'Bearer {TOKEN}', 'Content-Type': 'application/json'}

# Start cluster
def start_cluster(cluster_id):
    response = requests.post(
        f"{DATABRICKS_HOST}/api/2.1/clusters/start",
        headers=headers,
        json={"cluster_id": cluster_id}
    )
    return response.status_code == 200

# Stop cluster
def stop_cluster(cluster_id):
    response = requests.post(
        f"{DATABRICKS_HOST}/api/2.1/clusters/terminate", 
        headers=headers,
        json={"cluster_id": cluster_id}
    )
    return response.status_code == 200
