📁 File System Operations (dbutils.fs)

▼

Basic File Operations

# List files and directories in a path
files = dbutils.fs.ls("/mnt/data/")
for file in files:
    print(f"Name: {file.name}, Path: {file.path}, Size: {file.size}")

# List files recursively - useful for exploring nested directory structures
all_files = []
def list_recursive(path):
    """Recursively list all files in a directory structure"""
    for item in dbutils.fs.ls(path):
        if item.isDir():
            list_recursive(item.path)  # Recurse into subdirectories
        else:
            all_files.append(item.path)

list_recursive("/mnt/data/")
print(f"Found {len(all_files)} files total")

# Check if a file or directory exists
if dbutils.fs.ls("/mnt/data/important_file.csv"):
    print("File exists!")
else:
    print("File not found")

File Content Operations

# Read file content - useful for small files like config files
content = dbutils.fs.head("/mnt/data/config.txt", max_bytes=1000)
print("First 1000 bytes:", content)

# Copy files between locations
# Source can be local file system, DBFS, or cloud storage
dbutils.fs.cp("/mnt/source/data.csv", "/mnt/destination/data.csv")

# Copy entire directory recursively
dbutils.fs.cp("/mnt/source/folder/", "/mnt/destination/folder/", recurse=True)

# Move/rename files - atomically moves file to new location
dbutils.fs.mv("/mnt/old/location.csv", "/mnt/new/location.csv")

# Delete files - be careful with this operation!
dbutils.fs.rm("/mnt/data/temp_file.csv")

# Delete directory and all contents recursively
dbutils.fs.rm("/mnt/data/temp_directory/", recurse=True)

Working with Cloud Storage

# Mount cloud storage to DBFS for easy access
# Example: Mount S3 bucket with access keys
dbutils.fs.mount(
    source="s3a://my-bucket/data/",
    mount_point="/mnt/s3-data",
    extra_configs={
        "fs.s3a.access.key": dbutils.secrets.get("aws", "access-key"),
        "fs.s3a.secret.key": dbutils.secrets.get("aws", "secret-key")
    }
)

# Mount Azure Data Lake Storage Gen2
dbutils.fs.mount(
    source="abfss://container@storageaccount.dfs.core.windows.net/",
    mount_point="/mnt/adls",
    extra_configs={
        "fs.azure.account.auth.type.storageaccount.dfs.core.windows.net": "OAuth",
        "fs.azure.account.oauth.provider.type.storageaccount.dfs.core.windows.net": 
            "org.apache.hadoop.fs.azurebfs.oauth2.ClientCredsTokenProvider",
        "fs.azure.account.oauth2.client.id.storageaccount.dfs.core.windows.net": 
            dbutils.secrets.get("azure", "client-id"),
        "fs.azure.account.oauth2.client.secret.storageaccount.dfs.core.windows.net": 
            dbutils.secrets.get("azure", "client-secret"),
        "fs.azure.account.oauth2.client.endpoint.storageaccount.dfs.core.windows.net": 
            "https://login.microsoftonline.com/tenant-id/oauth2/token"
    }
)

# List all mounts to see what's currently mounted
mounts = dbutils.fs.mounts()
for mount in mounts:
    print(f"Mount point: {mount.mountPoint}, Source: {mount.source}")

# Unmount when no longer needed
dbutils.fs.unmount("/mnt/s3-data")

Working with Secret Scopes

# List all available secret scopes in your workspace
scopes = dbutils.secrets.listScopes()
print("Available secret scopes:")
for scope in scopes:
    print(f"  - {scope.name}")

# List secrets within a specific scope (names only, not values)
secrets = dbutils.secrets.list("my-secret-scope")
print("Secrets in 'my-secret-scope':")
for secret in secrets:
    print(f"  - {secret.key}")

# Retrieve secret value - this is the secure way to access credentials
api_key = dbutils.secrets.get("api-keys", "external-service-key")
database_password = dbutils.secrets.get("database", "production-password")

# Use secrets in connection strings
connection_string = f"""
jdbc:postgresql://prod-db.company.com:5432/analytics
?user=analytics_user
&password={dbutils.secrets.get("database", "analytics-password")}
"""

# Best practice: Use secrets in DataFrame operations
df = spark.read.format("jdbc") \
    .option("url", "jdbc:mysql://db.example.com:3306/mydb") \
    .option("dbtable", "users") \
    .option("user", "myuser") \
    .option("password", dbutils.secrets.get("mysql", "password")) \
    .load()

Creating Interactive Widgets

# Create different types of input widgets for parameterized notebooks

# Text widget - for free-form text input
dbutils.widgets.text("environment", "dev", "Environment")
env = dbutils.widgets.get("environment")
print(f"Running in environment: {env}")

# Dropdown widget - for selecting from predefined options
dbutils.widgets.dropdown("data_source", "customers", 
                         ["customers", "orders", "products", "inventory"],
                         "Choose Data Source")
selected_table = dbutils.widgets.get("data_source")

# Multiselect widget - for selecting multiple options
dbutils.widgets.multiselect("regions", "us-east-1", 
                           ["us-east-1", "us-west-2", "eu-west-1", "ap-south-1"],
                           "Select Regions")
selected_regions = dbutils.widgets.get("regions").split(",")

# Combobox widget - combines text input with dropdown suggestions
dbutils.widgets.combobox("date_range", "last_30_days",
                        ["last_7_days", "last_30_days", "last_90_days", "custom"],
                        "Date Range")
date_range = dbutils.widgets.get("date_range")

# Use widget values in your data processing logic
if env == "prod":
    table_prefix = "production"
    cluster_size = "large"
else:
    table_prefix = "staging" 
    cluster_size = "small"

# Dynamic table name based on widget selection
full_table_name = f"{table_prefix}.{selected_table}"
df = spark.table(full_table_name)

print(f"Processing table: {full_table_name} for regions: {selected_regions}")

# Remove widgets when done (useful for cleanup)
dbutils.widgets.remove("environment")
# Remove all widgets at once
dbutils.widgets.removeAll()

# Example of a complete ETL workflow using notebook orchestration
def run_etl_pipeline(date):
    """Run complete ETL pipeline by orchestrating multiple notebooks"""
    
    # Step 1: Data extraction
    extract_result = dbutils.notebook.run(
        "/etl/01_extract_data",
        timeout_seconds=1800,
        arguments={"processing_date": date}
    )
    
    if "ERROR" in extract_result:
        raise Exception(f"Extraction failed: {extract_result}")
    
    # Step 2: Data transformation
    transform_result = dbutils.notebook.run(
        "/etl/02_transform_data", 
        timeout_seconds=3600,
        arguments={"processing_date": date}
    )
    
    if "ERROR" in transform_result:
        raise Exception(f"Transformation failed: {transform_result}")
    
    # Step 3: Data loading/publishing
    load_result = dbutils.notebook.run(
        "/etl/03_load_data",
        timeout_seconds=1200, 
        arguments={"processing_date": date}
    )
    
    return load_result

# Run the pipeline
try:
    pipeline_result = run_etl_pipeline("2024-01-15")
    print(f"ETL Pipeline completed successfully: {pipeline_result}")
except Exception as e:
    print(f"Pipeline failed: {str(e)}")
    # Send alert or notification
    dbutils.notebook.exit(f"PIPELINE_FAILURE: {str(e)}")

Error Handling and Logging

import logging
from datetime import datetime

def safe_file_operation(operation, path, **kwargs):
    """Safely perform file operations with proper error handling"""
    try:
        if operation == "ls":
            return dbutils.fs.ls(path)
        elif operation == "rm":
            return dbutils.fs.rm(path, **kwargs)
        elif operation == "cp":
            return dbutils.fs.cp(path, kwargs.get("dest"), **kwargs)
        elif operation == "mv":
            return dbutils.fs.mv(path, kwargs.get("dest"))
    except Exception as e:
        logging.error(f"File operation {operation} failed for {path}: {str(e)}")
        return None

def create_checkpoint_file(checkpoint_path, status, metadata=None):
    """Create checkpoint files to track processing status"""
    checkpoint_data = {
        "timestamp": datetime.now().isoformat(),
        "status": status,
        "metadata": metadata or {}
    }
    
    # Convert to JSON and write to DBFS
    import json
    checkpoint_content = json.dumps(checkpoint_data, indent=2)
    
    # Write to a temp file first, then move (atomic operation)
    temp_path = f"{checkpoint_path}.tmp"
    dbutils.fs.put(temp_path, checkpoint_content, overwrite=True)
    dbutils.fs.mv(temp_path, checkpoint_path)

# Usage example
checkpoint_path = "/mnt/data/processing_checkpoints/job_2024_01_15.json"
create_checkpoint_file(checkpoint_path, "completed", 
                      {"records_processed": 1000000, "runtime_minutes": 45})

Production Best Practices

# Example of a robust data processing function using dbutils best practices
def process_daily_data(processing_date, source_path, target_path):
    """
    Process daily data with comprehensive error handling and monitoring
    
    Args:
        processing_date: Date to process in YYYY-MM-DD format
        source_path: Source data location
        target_path: Target output location
    """
    
    # Step 1: Validate inputs using dbutils
    try:
        files = dbutils.fs.ls(source_path)
        if not files:
            raise ValueError(f"No files found in source path: {source_path}")
    except Exception as e:
        dbutils.notebook.exit(f"ERROR: Source validation failed - {str(e)}")
    
    # Step 2: Use widgets for configuration
    dbutils.widgets.text("batch_size", "1000", "Batch Size")
    batch_size = int(dbutils.widgets.get("batch_size"))
    
    # Step 3: Secure credential access
    db_password = dbutils.secrets.get("production", "database-password")
    
    # Step 4: Create processing checkpoint
    checkpoint_dir = f"/mnt/checkpoints/daily_processing/{processing_date}"
    dbutils.fs.mkdirs(checkpoint_dir)
    
    try:
        # Your data processing logic here
        print(f"Processing {len(files)} files with batch size {batch_size}")
        
        # Mark as completed
        dbutils.fs.put(f"{checkpoint_dir}/SUCCESS", 
                       f"Completed at {datetime.now()}")
        
        return "SUCCESS"
        
    except Exception as e:
        # Mark as failed with error details
        dbutils.fs.put(f"{checkpoint_dir}/FAILED", 
                       f"Failed at {datetime.now()}: {str(e)}")
        raise e
