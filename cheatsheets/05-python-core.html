Basic File Reading and Writing

File I/O operations are fundamental for data engineering. Always use context managers (with statements) for automatic resource cleanup, handle encoding explicitly, and implement proper error handling for robust data processing pipelines.

import os
import json
from pathlib import Path

# Basic file reading - always use context managers for automatic cleanup
def read_text_file(file_path):
    """Read entire text file content safely"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            content = file.read()  # Read entire file at once
            return content
    except FileNotFoundError:
        print(f"File not found: {file_path}")
        return None
    except UnicodeDecodeError as e:
        print(f"Encoding error reading {file_path}: {e}")
        return None

# Read file line by line - memory efficient for large files
def read_lines(file_path):
    """Read file line by line to handle large files efficiently"""
    lines = []
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            for line_number, line in enumerate(file, 1):
                # Strip whitespace and skip empty lines
                cleaned_line = line.strip()
                if cleaned_line:  # Only keep non-empty lines
                    lines.append(cleaned_line)
                    
                # Optional: Log progress for very large files
                if line_number % 10000 == 0:
                    print(f"Processed {line_number} lines...")
    except Exception as e:
        print(f"Error reading file: {e}")
    
    return lines

# Writing files with proper error handling
def write_text_file(file_path, content, mode='w'):
    """Write content to file with comprehensive error handling"""
    try:
        # Create directory if it doesn't exist
        os.makedirs(os.path.dirname(file_path), exist_ok=True)
        
        with open(file_path, mode, encoding='utf-8') as file:
            if isinstance(content, list):
                # Write list of lines
                for line in content:
                    file.write(f"{line}\n")
            else:
                # Write string content
                file.write(content)
                
        print(f"Successfully wrote to {file_path}")
        return True
    except PermissionError:
        print(f"Permission denied writing to {file_path}")
        return False
    except IOError as e:
        print(f"IO Error: {e}")
        return False

# CSV file operations - essential for data processing
def read_csv_file(file_path, delimiter=','):
    """Read CSV file and return list of dictionaries"""
    data = []
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            # Use DictReader to automatically map columns to keys
            csv_reader = csv.DictReader(file, delimiter=delimiter)
            
            for row_number, row in enumerate(csv_reader, 1):
                # Clean up data - strip whitespace from all values
                cleaned_row = {key.strip(): value.strip() 
                              for key, value in row.items()}
                data.append(cleaned_row)
                
                # Optional: Validate data integrity
                if not cleaned_row.get('id'):
                    print(f"Warning: Row {row_number} missing ID")
                    
    except Exception as e:
        print(f"Error reading CSV: {e}")
    
    return data

def write_csv_file(file_path, data, fieldnames=None):
    """Write list of dictionaries to CSV file"""
    if not data:
        print("No data to write")
        return False
        
    try:
        # Auto-detect fieldnames if not provided
        if fieldnames is None:
            fieldnames = data[0].keys()
            
        with open(file_path, 'w', newline='', encoding='utf-8') as file:
            writer = csv.DictWriter(file, fieldnames=fieldnames)
            writer.writeheader()  # Write column headers
            writer.writerows(data)  # Write all data rows
            
        print(f"Successfully wrote {len(data)} rows to {file_path}")
        return True
    except Exception as e:
        print(f"Error writing CSV: {e}")
        return False

def read_json_file(file_path):
    """Read and parse JSON file with error handling"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            data = json.load(file)  # Parse JSON into Python objects

    except Exception as e:
        print(f"Error reading JSON: {e}")
        return None

def write_json_file(file_path, data, indent=2):

    try:
        with open(file_path, 'w', encoding='utf-8') as file:
            # Pretty print with indentation for readability

        print(f"Successfully wrote JSON to {file_path}")
        return True
    except Exception as e:
        print(f"Error writing JSON: {e}")
        return False

Advanced File Operations with pathlib

from pathlib import Path
import shutil
import glob
from datetime import datetime

# Modern file path handling with pathlib - preferred over os.path
def explore_directory(directory_path):
    """Explore directory structure and file types using pathlib"""
    path = Path(directory_path)
    
    if not path.exists():
        print(f"Directory {directory_path} does not exist")
        return
    
    print(f"Exploring directory: {path.absolute()}")
    
    # Count different file types
    file_stats = {}
    total_size = 0
    
    # Iterate through all files recursively
    for file_path in path.rglob('*'):  # ** means recursive
        if file_path.is_file():
            # Get file extension and size
            extension = file_path.suffix.lower() or 'no_extension'
            file_size = file_path.stat().st_size
            
            # Update statistics
            if extension not in file_stats:
                file_stats[extension] = {'count': 0, 'total_size': 0}
            
            file_stats[extension]['count'] += 1
            file_stats[extension]['total_size'] += file_size
            total_size += file_size
    
    # Print summary
    print(f"Total files: {sum(stats['count'] for stats in file_stats.values())}")
    print(f"Total size: {total_size:,} bytes ({total_size/1024/1024:.1f} MB)")
    
    for ext, stats in sorted(file_stats.items()):
        avg_size = stats['total_size'] / stats['count']
        print(f"  {ext}: {stats['count']} files, "
              f"avg size: {avg_size:,.0f} bytes")

def backup_files_by_pattern(source_dir, backup_dir, pattern="*.py"):
    """Backup files matching a pattern with timestamp"""
    source_path = Path(source_dir)
    backup_path = Path(backup_dir)
    
    # Create backup directory with timestamp
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    timestamped_backup = backup_path / f"backup_{timestamp}"
    timestamped_backup.mkdir(parents=True, exist_ok=True)
    
    # Find and copy matching files
    copied_count = 0
    for file_path in source_path.rglob(pattern):
        if file_path.is_file():
            # Preserve directory structure in backup
            relative_path = file_path.relative_to(source_path)
            backup_file_path = timestamped_backup / relative_path
            
            # Create parent directories if needed
            backup_file_path.parent.mkdir(parents=True, exist_ok=True)
            
            # Copy the file
            shutil.copy2(file_path, backup_file_path)
            copied_count += 1
    
    print(f"Backed up {copied_count} files to {timestamped_backup}")
    return timestamped_backup

# File monitoring and cleanup functions
def cleanup_old_files(directory, days_old=30, dry_run=True):
    """Clean up files older than specified days"""
    path = Path(directory)
    cutoff_time = datetime.now().timestamp() - (days_old * 24 * 60 * 60)
    
    files_to_delete = []
    total_size_saved = 0
    
    for file_path in path.rglob('*'):
        if file_path.is_file():
            # Check file modification time
            if file_path.stat().st_mtime < cutoff_time:
                file_size = file_path.stat().st_size
                files_to_delete.append((file_path, file_size))
                total_size_saved += file_size
    
    print(f"Found {len(files_to_delete)} files older than {days_old} days")
    print(f"Total size to free: {total_size_saved:,} bytes "
          f"({total_size_saved/1024/1024:.1f} MB)")
    
    if not dry_run and files_to_delete:
        for file_path, _ in files_to_delete:
            try:
                file_path.unlink()  # Delete the file
                print(f"Deleted: {file_path}")
            except Exception as e:
                print(f"Error deleting {file_path}: {e}")
    elif dry_run:
        print("This was a dry run. Use dry_run=False to actually delete files.")
    
    return files_to_delete

# Directory operations with error handling
def create_directory_structure(base_path, structure):
    """
    Create a complete directory structure from a nested dictionary
    
    Args:
        base_path: Root directory path
        structure: Dict representing directory structure
        
    Example:
        structure = {
            'data': {
                'raw': {},
                'processed': {},
                'archive': {}
            },
            'logs': {},
            'config': {}
        }
    """
    def create_recursive(current_path, struct):
        for name, children in struct.items():
            dir_path = current_path / name
            dir_path.mkdir(exist_ok=True)
            print(f"Created directory: {dir_path}")
            
            # If children is a dict, recurse
            if isinstance(children, dict):
                create_recursive(dir_path, children)
    
    base = Path(base_path)
    create_recursive(base, structure)

# Running external commands safely
def run_command(command, cwd=None, timeout=60):
    """
    Run external command safely with proper error handling
    
    Args:
        command: Command to run (string or list)
        cwd: Working directory for command
        timeout: Timeout in seconds
    """
    try:
        # Use shell=False for security when command is a list
        if isinstance(command, str):
            # Split string command into list for safer execution
            command = command.split()
        
        result = subprocess.run(
            command,
            cwd=cwd,
            capture_output=True,
            text=True,
            timeout=timeout,
            check=True  # Raises exception on non-zero exit code
        )
        
        return {
            'success': True,
            'stdout': result.stdout,
            'stderr': result.stderr,
            'return_code': result.returncode
        }
        
    except subprocess.TimeoutExpired:
        return {'success': False, 'error': f'Command timed out after {timeout}s'}
    except subprocess.CalledProcessError as e:
        return {
            'success': False, 
            'error': f'Command failed with code {e.returncode}',
            'stdout': e.stdout,
            'stderr': e.stderr
        }
    except Exception as e:
        return {'success': False, 'error': str(e)}

# Generator expression - memory efficient for large datasets
def process_large_dataset(filename):
    """Process large file line by line without loading into memory"""
    with open(filename, 'r') as file:
        # Generator expression - processes one line at a time
        processed_lines = (
            line.strip().upper() 
            for line in file 
            if line.strip()  # Skip empty lines
        )
        
        # Process each line individually
        for processed_line in processed_lines:
            yield processed_line  # Yield instead of storing in memory

def safe_divide(a, b):
    """Demonstrate comprehensive error handling"""
    try:
        result = a / b
        return {'success': True, 'result': result}
    except ZeroDivisionError:
        logging.error(f"Division by zero: {a} / {b}")
        return {'success': False, 'error': 'Division by zero'}
    except TypeError as e:
        logging.error(f"Type error in division: {e}")
        return {'success': False, 'error': f'Invalid types: {type(a)}, {type(b)}'}
    except Exception as e:
        logging.error(f"Unexpected error in division: {e}")
        return {'success': False, 'error': 'Unexpected error occurred'}
    finally:
        # This always executes - useful for cleanup
        logging.debug(f"Division operation completed for {a}, {b}")

# Custom context manager for timing operations
@contextmanager
def timer(operation_name):
    """Context manager to time operations"""
    start_time = time.time()
    try:
        print(f"Starting {operation_name}...")
        yield  # This is where the with-block executes
    except Exception as e:
        print(f"Error during {operation_name}: {e}")
        raise  # Re-raise the exception
    finally:
        end_time = time.time()
        duration = end_time - start_time
        print(f"{operation_name} completed in {duration:.2f} seconds")

import functools
import time
from datetime import datetime
import logging

# Function decorator for logging execution time and parameters
def log_execution(func):
    """Decorator to log function execution details"""
    @functools.wraps(func)  # Preserves original function metadata
    def wrapper(*args, **kwargs):
        # Log function call with parameters
        logging.info(f"Calling {func.__name__} with args={args}, kwargs={kwargs}")
        
        start_time = time.time()
        try:
            result = func(*args, **kwargs)
            execution_time = time.time() - start_time
            logging.info(f"{func.__name__} completed successfully in {execution_time:.3f}s")
            return result
        except Exception as e:
            execution_time = time.time() - start_time
            logging.error(f"{func.__name__} failed after {execution_time:.3f}s: {e}")
            raise
    return wrapper

# Retry decorator for handling transient failures
def retry(max_attempts=3, delay=1, backoff=2):
    """Decorator to retry function calls with exponential backoff"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            attempts = 0
            current_delay = delay
            
            while attempts < max_attempts:
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    attempts += 1
                    if attempts == max_attempts:
                        logging.error(f"{func.__name__} failed after {max_attempts} attempts")
                        raise
                    
                    logging.warning(f"{func.__name__} attempt {attempts} failed: {e}")
                    logging.info(f"Retrying in {current_delay} seconds...")
                    time.sleep(current_delay)
                    current_delay *= backoff
                    
        return wrapper
    return decorator

# Cache decorator for expensive computations
def memoize(func):
    """Decorator to cache function results"""
    cache = {}
    
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        # Create cache key from arguments
        cache_key = str(args) + str(sorted(kwargs.items()))
        
        if cache_key in cache:
            logging.debug(f"Cache hit for {func.__name__}")
            return cache[cache_key]
        
        logging.debug(f"Cache miss for {func.__name__}, computing...")
        result = func(*args, **kwargs)
        cache[cache_key] = result
        return result
    
    # Add cache inspection methods
    wrapper.cache = cache
    wrapper.cache_info = lambda: {
        'size': len(cache),
        'hits': getattr(wrapper, '_hits', 0),
        'misses': getattr(wrapper, '_misses', 0)
    }
    
    return wrapper

# Property decorators for class methods
class DataProcessor:
    def __init__(self, data):
        self._data = data
        self._processed = None
    
    @property
    def data(self):
        """Getter for data"""
        return self._data
    
    @data.setter
    def data(self, value):
        """Setter for data - triggers reprocessing"""
        self._data = value
        self._processed = None  # Reset processed data
    
    @property
    def processed_data(self):
        """Lazy loading - only process when needed"""
        if self._processed is None:
            print("Processing data...")
            self._processed = [x * 2 for x in self._data]
        return self._processed
    
    @staticmethod
    def validate_data(data):
        """Static method - doesn't need instance"""
        return all(isinstance(x, (int, float)) for x in data)
    
    @classmethod
    def from_file(cls, filename):
        """Alternative constructor - creates instance from file"""
        with open(filename, 'r') as f:
            data = [float(line.strip()) for line in f]
        return cls(data)
