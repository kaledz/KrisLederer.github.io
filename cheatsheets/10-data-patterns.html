Lambda Architecture

# Lambda Architecture Implementation
class LambdaArchitecture:
    def __init__(self):
        self.batch_layer = BatchLayer()
        self.speed_layer = SpeedLayer()
        self.serving_layer = ServingLayer()
    
    def process_data(self, raw_data):
        # Batch Layer: Complete, accurate views from immutable data
        batch_view = self.batch_layer.process(raw_data)
        
        # Speed Layer: Fast, approximate views for recent data
        realtime_view = self.speed_layer.process(raw_data)
        
        # Serving Layer: Merge batch and speed views
        unified_view = self.serving_layer.merge(batch_view, realtime_view)
        
        return unified_view

class BatchLayer:
    """Handles large-scale batch processing"""
    def __init__(self):
        self.storage = "HDFS/S3"  # Immutable storage
        self.compute = "Spark/MapReduce"
    
    def process(self, data):
        # Process complete dataset periodically
        return self.compute_batch_views(data)
    
    def compute_batch_views(self, data):
        """Example: Daily aggregations"""
        daily_stats = (
            spark.read.parquet(data)
            .groupBy("date", "user_id")
            .agg(
                count("events").alias("event_count"),
                sum("revenue").alias("total_revenue")
            )
        )
        return daily_stats

Kappa Architecture (Stream-First)

class KappaArchitecture:
    """Simplified architecture using only stream processing"""
    
    def __init__(self):
        self.stream_processor = StreamProcessor()
        self.state_store = StateStore()
    
    def process_event(self, event):
        # Single stream processing pipeline handles all data
        processed_event = self.stream_processor.process(event)
        
        # Update materialized views
        self.update_views(processed_event)
        
        return processed_event
    
    def reprocess_historical_data(self, start_time, end_time):
        """Reprocess data by replaying streams"""
        # Reset state to start_time
        self.state_store.reset_to_timestamp(start_time)
        
        # Replay events from Kafka
        events = self.replay_events(start_time, end_time)
        
        for event in events:
            self.process_event(event)

Medallion Architecture (Bronze-Silver-Gold)

# Medallion Architecture with Delta Lake
import delta
from pyspark.sql import SparkSession
from pyspark.sql.functions import *

class MedallionPipeline:
    def __init__(self, spark):
        self.spark = spark
        self.bronze_path = "/delta/bronze"
        self.silver_path = "/delta/silver" 
        self.gold_path = "/delta/gold"
    
    def bronze_layer(self, raw_data_path):
        """Bronze: Raw data ingestion with minimal transformation"""
        bronze_df = (
            self.spark.read
            .option("multiline", "true")
            .json(raw_data_path)
            .withColumn("_ingestion_timestamp", current_timestamp())
            .withColumn("_source_file", input_file_name())
        )
        
        # Write to Bronze Delta table
        (
            bronze_df.write
            .format("delta")
            .mode("append")
            .option("mergeSchema", "true")  # Handle schema evolution
            .save(self.bronze_path)
        )
        
        return bronze_df
    
    def silver_layer(self):
        """Silver: Cleaned, validated, deduplicated data"""
        bronze_df = self.spark.read.format("delta").load(self.bronze_path)
        
        silver_df = (
            bronze_df
            # Data quality checks
            .filter(col("user_id").isNotNull())
            .filter(col("event_timestamp").isNotNull())
            
            # Data standardization
            .withColumn("event_timestamp", 
                       to_timestamp(col("event_timestamp"), "yyyy-MM-dd HH:mm:ss"))
            .withColumn("user_id", upper(trim(col("user_id"))))
            
            # Deduplication
            .dropDuplicates(["user_id", "event_timestamp", "event_type"])
        )
        
        return silver_df

Incremental Processing Pattern

class IncrementalProcessor:
    def __init__(self, spark, table_name, watermark_column="updated_at"):
        self.spark = spark
        self.table_name = table_name
        self.watermark_column = watermark_column
        self.checkpoint_table = f"{table_name}_checkpoint"
    
    def get_last_processed_timestamp(self):
        """Get the last processed timestamp from checkpoint"""
        try:
            checkpoint_df = self.spark.table(self.checkpoint_table)
            last_timestamp = checkpoint_df.agg(max("last_processed_timestamp")).collect()[0][0]
            return last_timestamp
        except:
            # First run - start from beginning or specific date
            return "1900-01-01 00:00:00"
    
    def process_incremental_data(self, source_table):
        """Process only new/updated data since last checkpoint"""
        last_processed = self.get_last_processed_timestamp()
        
        # Read only incremental data
        incremental_df = (
            self.spark.table(source_table)
            .filter(col(self.watermark_column) > last_processed)
        )
        
        if incremental_df.count() == 0:
            print("No new data to process")
            return None
        
        # Process the incremental data
        processed_df = self.transform_data(incremental_df)
        
        # Update checkpoint
        new_watermark = incremental_df.agg(max(self.watermark_column)).collect()[0][0]
        self.update_checkpoint(new_watermark)
        
        return processed_df

from pyspark.sql.functions import *
from pyspark.sql.types import *

class DataQualityFramework:
    def __init__(self, spark):
        self.spark = spark
        self.quality_rules = []
        self.results = []
    
    def add_rule(self, rule_name, condition, action="flag"):
        """Add data quality rule"""
        self.quality_rules.append({
            "name": rule_name,
            "condition": condition,
            "action": action
        })
    
    def validate_data(self, df, table_name="unknown"):
        """Apply all quality rules to dataframe"""
        results = []
        
        for rule in self.quality_rules:
            result = self.apply_rule(df, rule, table_name)
            results.append(result)
        
        return results
    
    def apply_rule(self, df, rule, table_name):
        """Apply single quality rule"""
        condition = rule["condition"]
        
        # Count violations
        violations_df = df.filter(~condition)  # NOT condition
        violation_count = violations_df.count()
        total_count = df.count()
        
        result = {
            "table_name": table_name,
            "rule_name": rule["name"],
            "total_records": total_count,
            "violations": violation_count,
            "violation_rate": violation_count / total_count if total_count > 0 else 0,
            "timestamp": current_timestamp(),
            "action": rule["action"]
        }
        
        # Take action based on rule
        if rule["action"] == "fail" and violation_count > 0:
            raise DataQualityException(f"Quality rule failed: {rule['name']}")
        elif rule["action"] == "quarantine" and violation_count > 0:
            self.quarantine_bad_data(violations_df, rule["name"])
        
        return result

# Usage example
dq = DataQualityFramework(spark)

# Define quality rules
dq.add_rule("not_null_user_id", col("user_id").isNotNull(), "fail")
dq.add_rule("valid_email_format", col("email").rlike(r"^[^@]+@[^@]+\.[^@]+$"), "quarantine")
dq.add_rule("reasonable_age", col("age").between(0, 150), "flag")

Pipeline Monitoring

import logging
import time
from datetime import datetime
from typing import Dict, Any

class PipelineMonitor:
    def __init__(self, pipeline_name: str):
        self.pipeline_name = pipeline_name
        self.metrics = {}
        self.alerts = []
        
        # Setup logging
        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(f"pipeline.{pipeline_name}")
    
    def track_execution(self, stage_name: str):
        """Decorator to track stage execution"""
        def decorator(func):
            def wrapper(*args, **kwargs):
                start_time = time.time()
                stage_id = f"{self.pipeline_name}.{stage_name}"
                
                self.logger.info(f"Starting stage: {stage_name}")
                
                try:
                    result = func(*args, **kwargs)
                    
                    # Track success metrics
                    execution_time = time.time() - start_time
                    self.record_metric(stage_id, "execution_time", execution_time)
                    self.record_metric(stage_id, "status", "success")
                    
                    if hasattr(result, 'count'):
                        record_count = result.count()
                        self.record_metric(stage_id, "record_count", record_count)
                        self.logger.info(f"Stage {stage_name} processed {record_count} records in {execution_time:.2f}s")
                    
                    return result
                    
                except Exception as e:
                    # Track failure metrics
                    self.record_metric(stage_id, "status", "failed")
                    self.record_metric(stage_id, "error", str(e))
                    self.logger.error(f"Stage {stage_name} failed: {str(e)}")
                    
                    # Send alert
                    self.send_alert(f"Pipeline stage failed: {stage_name}", str(e))
                    raise
                    
            return wrapper
        return decorator

Circuit Breaker Pattern

import time
from enum import Enum
from datetime import datetime, timedelta

class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Circuit is open, failing fast
    HALF_OPEN = "half_open"  # Testing if service is back

class CircuitBreaker:
    def __init__(self, 
                 failure_threshold: int = 5,
                 timeout: int = 60,
                 expected_exception: type = Exception):
        
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.expected_exception = expected_exception
        
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
    
    def call(self, func, *args, **kwargs):
        """Execute function with circuit breaker protection"""
        
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
            else:
                raise CircuitBreakerOpenError("Circuit breaker is OPEN")
        
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
            
        except self.expected_exception as e:
            self._on_failure()
            raise e
    
    def _on_failure(self):
        """Handle failed execution"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN

Retry with Exponential Backoff

import random
import time
from functools import wraps

def retry_with_backoff(max_retries: int = 3, 
                      base_delay: float = 1.0,
                      max_delay: float = 60.0,
                      exponential_base: float = 2.0,
                      jitter: bool = True):
    """Decorator for retrying functions with exponential backoff"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                    
                except Exception as e:
                    last_exception = e
                    
                    if attempt == max_retries:
                        # Last attempt, re-raise exception
                        raise e
                    
                    # Calculate delay
                    delay = min(
                        base_delay * (exponential_base ** attempt),
                        max_delay
                    )
                    
                    # Add jitter to avoid thundering herd
                    if jitter:
                        delay = delay * (0.5 + random.random() * 0.5)
                    
                    print(f"Attempt {attempt + 1} failed: {e}. Retrying in {delay:.2f}s...")
                    time.sleep(delay)
            
            raise last_exception
        
        return wrapper
    return decorator

# Usage example
@retry_with_backoff(max_retries=3, base_delay=1.0, max_delay=30.0)
def unreliable_operation():
    """Simulate an unreliable operation"""
    if random.random() < 0.7:  # 70% failure rate
        raise ConnectionError("Service temporarily unavailable")
    
    return {"status": "success"}

Caching Strategies

import time
from functools import wraps
from typing import Any, Dict, Callable

class CacheManager:
    def __init__(self):
        self.memory_cache = {}
        self.cache_stats = {
            "hits": 0,
            "misses": 0,
            "evictions": 0
        }
    
    def cache_with_ttl(self, ttl_seconds: int = 300):
        """Cache decorator with time-to-live"""
        def decorator(func: Callable) -> Callable:
            @wraps(func)
            def wrapper(*args, **kwargs):
                # Create cache key
                cache_key = self._create_cache_key(func.__name__, args, kwargs)
                
                # Check cache
                cached_result = self._get_from_cache(cache_key)
                if cached_result is not None:
                    self.cache_stats["hits"] += 1
                    return cached_result
                
                # Cache miss - execute function
                self.cache_stats["misses"] += 1
                result = func(*args, **kwargs)
                
                # Store in cache
                self._put_in_cache(cache_key, result, ttl_seconds)
                
                return result
            
            return wrapper
        return decorator
    
    def get_cache_stats(self) -> Dict:
        """Get cache statistics"""
        total_requests = self.cache_stats["hits"] + self.cache_stats["misses"]
        hit_rate = self.cache_stats["hits"] / total_requests if total_requests > 0 else 0
        
        return {
            **self.cache_stats,
            "hit_rate": hit_rate,
            "cache_size": len(self.memory_cache)
        }

    • Cost per pipeline run
    • Resource utilization
