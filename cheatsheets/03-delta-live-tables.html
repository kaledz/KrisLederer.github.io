Basic DLT Decorators

import dlt
from pyspark.sql.functions import *
from pyspark.sql.types import *

# Simple table
@dlt.table(
    comment="Raw customer data from source system",
    table_properties={
        "quality": "bronze",
        "pipelines.autoOptimize.managed": "true"
    }
)
def raw_customers():
    return spark.read.json("/path/to/customer/data")

# View (not materialized)
@dlt.view(
    comment="Filtered view of active customers"
)
def active_customers():
    return dlt.read("raw_customers").filter(col("status") == "active")

# Streaming table
@dlt.table(
    comment="Streaming orders from Kafka"
)
def streaming_orders():
    return (
        spark.readStream
        .format("kafka")
        .option("kafka.bootstrap.servers", "localhost:9092")
        .option("subscribe", "orders")
        .load()
    )

⚡ AWS Kinesis Integration

Basic Kinesis Stream Reading

@dlt.table(
    comment="Bronze layer - Raw data from Kinesis stream",
    table_properties={
        "quality": "bronze",
        "pipelines.trigger.interval": "10 seconds"
    }
)
def kinesis_bronze():
    return (
        spark.readStream
        .format("kinesis")
        .option("streamName", "user-events-stream")
        .option("region", "us-west-2")
        .option("initialPosition", "TRIM_HORIZON")  # or "LATEST"
        .option("awsAccessKeyId", dbutils.secrets.get("aws", "access-key"))
        .option("awsSecretKey", dbutils.secrets.get("aws", "secret-key"))
        .load()
        .withColumn("_ingest_timestamp", current_timestamp())
        .withColumn("_partition_key", col("partitionKey"))
        .withColumn("_sequence_number", col("sequenceNumber"))
    )

# With IAM role (recommended for production)
@dlt.table(
    comment="Kinesis stream with IAM role authentication"
)
def kinesis_bronze_iam():
    return (
        spark.readStream
        .format("kinesis")
        .option("streamName", "production-events")
        .option("region", "us-east-1")
        .option("initialPosition", "LATEST")
        .option("roleArn", "arn:aws:iam::123456789012:role/DatabricksKinesisRole")
        .option("roleSessionName", "DLT-Kinesis-Session")
        .load()
    )

Kinesis with Custom Deserialization

# Define schema for JSON payloads
kinesis_schema = StructType([
    StructField("event_id", StringType(), True),
    StructField("user_id", StringType(), True),
    StructField("event_type", StringType(), True),
    StructField("timestamp", TimestampType(), True),
    StructField("properties", MapType(StringType(), StringType()), True)
])

@dlt.table(
    comment="Parsed Kinesis events with schema validation"
)
def kinesis_parsed():
    return (
        dlt.read_stream("kinesis_bronze")
        .select(
            from_json(col("data").cast("string"), kinesis_schema).alias("event"),
            col("_ingest_timestamp"),
            col("_partition_key"),
            col("_sequence_number")
        )
        .select("event.*", "_ingest_timestamp", "_partition_key", "_sequence_number")
        .filter(col("event_id").isNotNull())  # Basic data quality check
    )

🏛️ Bronze-Silver-Gold Architecture

▼

Bronze Layer (Raw Data)

@dlt.table(
    comment="Bronze: Raw ingestion with minimal processing",
    table_properties={"quality": "bronze"}
)
def bronze_orders():
    return (
        spark.readStream
        .format("cloudFiles")
        .option("cloudFiles.format", "json")
        .option("cloudFiles.schemaLocation", "/tmp/bronze_schema/orders")
        .load("/mnt/raw/orders/")
        .withColumn("_bronze_ingest_timestamp", current_timestamp())
        .withColumn("_source_file", input_file_name())
    )

Silver Layer (Cleaned and Validated)

@dlt.table(
    comment="Silver: Cleaned and validated orders",
    table_properties={"quality": "silver"}
)
def silver_orders():
    return (
        dlt.read_stream("bronze_orders")
        .select(from_json(col("value"), orders_schema).alias("data"))
        .select("data.*", "_bronze_ingest_timestamp")
        .filter(col("order_id").isNotNull())
        .filter(col("customer_id").isNotNull())
        .filter(col("order_amount") > 0)
        .withColumn("order_date", to_date(col("order_timestamp")))
        .withColumn("_silver_processed_timestamp", current_timestamp())
        .dropDuplicates(["order_id"])  # Remove duplicates
    )

Gold Layer (Business Logic)

@dlt.table(
    comment="Gold: Daily customer order summary",
    table_properties={
        "quality": "gold",
        "business_domain": "sales"
    }
)
def gold_daily_customer_summary():
    return (
        dlt.read_stream("silver_orders")
        .join(dlt.read("silver_customers"), ["customer_id"], "inner")
        .groupBy("customer_id", "customer_tier", "order_date")
        .agg(
            count("order_id").alias("order_count"),
            sum("order_amount").alias("total_amount"),
            avg("order_amount").alias("avg_order_value"),
            max("order_timestamp").alias("last_order_time")
        )
        .withColumn("_gold_processed_timestamp", current_timestamp())
    )

✅ Data Quality and Expectations

▼

Basic Data Quality Checks

@dlt.table(
    comment="Orders with data quality validations"
)
@dlt.expect_or_fail("valid_order_id", "order_id IS NOT NULL")
@dlt.expect_or_fail("positive_amount", "order_amount > 0")
@dlt.expect_or_drop("valid_email", "email RLIKE '^[^@]+@[^@]+\\\\.[^@]+$'")
def validated_orders():
    return dlt.read("bronze_orders")

# Custom expectations
@dlt.table(comment="Customers with comprehensive validation")
@dlt.expect_or_fail("customer_id_not_null", "customer_id IS NOT NULL")
@dlt.expect_or_drop("valid_customer_email", "email IS NOT NULL AND email != ''")
@dlt.expect("reasonable_age", "age BETWEEN 13 AND 120")  # Allow but track violations
def validated_customers():
    return (
        dlt.read("bronze_customers")
        .withColumn("age", col("age").cast("int"))
        .filter(col("created_date") <= current_date())
    )

🌊 Streaming Tables and Auto Loader

Auto Loader for File Ingestion

@dlt.table(
    comment="Bronze layer - Auto Loader for CSV files"
)
def bronze_sales():
    return (
        spark.readStream
        .format("cloudFiles")
        .option("cloudFiles.format", "csv")
        .option("cloudFiles.schemaLocation", "/tmp/schema/sales")
        .option("cloudFiles.inferColumnTypes", "true")
        .option("cloudFiles.schemaEvolutionMode", "rescue")  # Handle schema changes
        .option("header", "true")
        .load("/mnt/data/sales/")
        .withColumn("_ingest_timestamp", current_timestamp())
        .withColumn("_source_file", input_file_name())
    )

# JSON Auto Loader
@dlt.table(
    comment="Auto Loader for JSON with schema evolution"
)
def bronze_events():
    return (
        spark.readStream
        .format("cloudFiles")
        .option("cloudFiles.format", "json")
        .option("cloudFiles.schemaLocation", "/tmp/schema/events")
        .option("cloudFiles.schemaEvolutionMode", "addNewColumns")
        .option("multiLine", "true")
        .load("/mnt/data/events/")
        .withColumn("_ingest_date", to_date(current_timestamp()))
    )

🔄 Change Data Capture (CDC) Patterns

▼ 
@dlt.table(comment="Process CDC events from source system")
def customers_cdc():
    return (
        spark.readStream
        .format("cloudFiles")
        .option("cloudFiles.format", "json")
        .option("cloudFiles.schemaLocation", "/tmp/schema/cdc")
        .load("/mnt/cdc/customers/")
        .select(
            col("data.*"),
            col("_change_type"),
            col("_commit_version"),
            col("_commit_timestamp")
        )
    )

# Apply CDC changes
@dlt.table(
    comment="Current state after applying CDC"
)
def customers_current():
    cdc_df = dlt.read_stream("customers_cdc")
    
    # Get latest version for each customer
    window_spec = Window.partitionBy("customer_id").orderBy(desc("_commit_timestamp"))
    
    return (
        cdc_df
        .withColumn("row_num", row_number().over(window_spec))
        .filter(col("row_num") == 1)
        .filter(col("_change_type") != "delete")
        .drop("row_num", "_change_type", "_commit_version", "_commit_timestamp")
    )
