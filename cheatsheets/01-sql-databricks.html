SELECT Basics

The SELECT statement forms the foundation of all SQL queries. Use column aliases for readability, apply transformations with functions, and always include appropriate filtering and ordering for performance.

-- Basic SELECT with column aliases for better readability
-- This query retrieves customer information with renamed columns
SELECT 
    customer_id AS id,              -- Rename for shorter reference
    customer_name AS name,          -- More user-friendly column name  
    UPPER(email) AS email_upper,    -- Transform email to uppercase for consistency
    created_date                    -- Keep original column name
FROM customers                      -- Source table
WHERE created_date >= '2023-01-01' -- Filter for records from 2023 onwards only
ORDER BY created_date DESC          -- Sort by newest records first for relevance
LIMIT 100;                          -- Restrict to first 100 results for performance

-- SELECT DISTINCT removes duplicate combinations of values
-- Use when you need unique combinations of columns
SELECT DISTINCT region, country     -- Get unique region-country pairs
FROM customers                      -- From the customers table
ORDER BY region, country;           -- Sort alphabetically for easy reading

WHERE Clause Patterns

The WHERE clause filters data before processing, which is crucial for performance. Use appropriate operators and handle NULL values explicitly to avoid unexpected results.

-- Multiple conditions with logical operators
-- Combine conditions using AND, OR, and parentheses for complex logic
SELECT * FROM orders 
WHERE order_date BETWEEN '2023-01-01' AND '2023-12-31'  -- Date range filter
  AND status IN ('completed', 'shipped')                -- Multiple status values
  AND amount > 100;                                     -- Minimum amount threshold

-- Pattern matching with LIKE and RLIKE (regex)
-- Use for flexible text searching and validation
SELECT * FROM customers
WHERE email LIKE '%@gmail.com'                          -- Ends with @gmail.com
  AND customer_name NOT LIKE 'Test%'                    -- Exclude test accounts
  AND phone RLIKE '^\\+1[0-9]{10}$';                   -- US phone number pattern

-- NULL handling - explicitly check for NULL values
-- NULL comparisons require special operators (IS NULL/IS NOT NULL)
SELECT * FROM customers
WHERE email IS NOT NULL                                 -- Must have email
  AND last_login IS NULL                                -- Never logged in
  AND COALESCE(phone, 'N/A') != 'N/A';                 -- Has phone number

CASE WHEN Statements

CASE WHEN statements provide conditional logic in SQL, similar to if-else statements in programming languages. Use them for data categorization, conditional calculations, and handling complex business logic.

-- Simple CASE statement for data categorization
-- Evaluates conditions in order and returns first matching result
SELECT 
    customer_id,
    CASE 
        WHEN age < 25 THEN 'Young'              -- First condition checked
        WHEN age BETWEEN 25 AND 54 THEN 'Adult' -- Second condition if first fails
        WHEN age >= 55 THEN 'Senior'            -- Third condition
        ELSE 'Unknown'                          -- Default if no conditions match
    END AS age_group,
    
    -- Nested CASE for complex hierarchical logic
    -- Inner CASE only evaluated when outer condition is true
    CASE 
        WHEN region = 'US' THEN 
            CASE WHEN state IN ('CA', 'NY', 'FL') THEN 'Major State'
                 ELSE 'Other State' 
            END
        WHEN region = 'EU' THEN 'Europe'        -- Simple assignment for EU
        ELSE 'Other Region'                     -- Default for all other regions
    END AS region_category
FROM customers;

All JOIN Types

-- INNER JOIN (most common)
SELECT 
    c.customer_id,
    c.customer_name,
    o.order_id,
    o.order_amount
FROM customers c
INNER JOIN orders o ON c.customer_id = o.customer_id
WHERE o.order_date >= '2023-01-01';

-- LEFT JOIN (keep all customers)
SELECT 
    c.customer_id,
    c.customer_name,
    COALESCE(SUM(o.order_amount), 0) AS total_orders
FROM customers c
LEFT JOIN orders o ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.customer_name;

Basic Aggregations

-- Standard aggregations
SELECT 
    region,
    COUNT(*) AS customer_count,
    COUNT(DISTINCT customer_id) AS unique_customers,
    AVG(order_amount) AS avg_order,
    SUM(order_amount) AS total_revenue,
    MIN(order_date) AS first_order,
    MAX(order_date) AS last_order,
    STDDEV(order_amount) AS revenue_std
FROM orders o
JOIN customers c ON o.customer_id = c.customer_id
GROUP BY region
HAVING COUNT(*) > 100
ORDER BY total_revenue DESC;

🪟 Window Functions

▼

ROW_NUMBER, RANK, DENSE_RANK

SELECT 
    customer_id,
    order_date,
    order_amount,
    
    -- Row number (unique sequential)
    ROW_NUMBER() OVER (
        PARTITION BY customer_id 
        ORDER BY order_date DESC
    ) AS order_sequence,
    
    -- Rank (gaps for ties)
    RANK() OVER (
        ORDER BY order_amount DESC
    ) AS amount_rank,
    
    -- Dense rank (no gaps)
    DENSE_RANK() OVER (
        PARTITION BY YEAR(order_date) 
        ORDER BY order_amount DESC
    ) AS yearly_dense_rank
FROM orders;

MERGE (Upsert Operations)

-- Basic MERGE (upsert)
MERGE INTO customer_profiles AS target
USING customer_updates AS source
ON target.customer_id = source.customer_id
WHEN MATCHED THEN
    UPDATE SET 
        email = source.email,
        phone = source.phone,
        last_updated = CURRENT_TIMESTAMP()
WHEN NOT MATCHED THEN
    INSERT (customer_id, email, phone, created_date)
    VALUES (source.customer_id, source.email, source.phone, CURRENT_TIMESTAMP());

Delta Lake Utilities

-- Optimize table (compaction)
OPTIMIZE customers
WHERE created_date >= '2023-01-01';

-- Optimize with Z-ordering
OPTIMIZE sales
ZORDER BY (customer_id, order_date);

-- Vacuum (remove old files)
VACUUM customers RETAIN 168 HOURS;  -- 7 days

-- Show table history
DESCRIBE HISTORY customers;

-- Time travel queries
SELECT * FROM customers VERSION AS OF 1;
SELECT * FROM customers TIMESTAMP AS OF '2023-12-01 00:00:00';

Query Optimization

-- Use column pruning
SELECT customer_id, order_amount  -- Only select needed columns
FROM orders
WHERE order_date >= '2023-01-01';

-- Predicate pushdown
SELECT *
FROM orders
WHERE partition_date = '2023-12-01'  -- Filter on partition column first
  AND order_amount > 100;

-- Use broadcast joins for small tables
SELECT /*+ BROADCAST(small_table) */ *
FROM large_table l
JOIN small_table s ON l.key = s.key;

ANSI Standard Data Types

-- ANSI SQL Standard Data Types
CREATE TABLE example_table (
    -- Numeric types (ANSI standard)
    id INTEGER,
    price DECIMAL(10,2),
    quantity NUMERIC(8,0),
    percentage REAL,
    rating DOUBLE PRECISION,
    
    -- Character types (ANSI standard)
    name VARCHAR(255),
    description TEXT,
    status CHAR(1),
    
    -- Date/Time types (ANSI standard)
    created_date DATE,
    updated_time TIMESTAMP,
    
    -- Boolean type (ANSI SQL:1999+)
    is_active BOOLEAN
);

-- ANSI compliant constraints
ALTER TABLE example_table
ADD CONSTRAINT pk_example PRIMARY KEY (id),
ADD CONSTRAINT chk_status CHECK (status IN ('A', 'I', 'P')),
ADD CONSTRAINT nn_name CHECK (name IS NOT NULL);

ANSI Standard Functions

-- String functions (ANSI SQL:2016)
SELECT 
    UPPER(name) as upper_name,                    -- Convert to uppercase
    LOWER(email) as lower_email,                  -- Convert to lowercase  
    LENGTH(description) as desc_length,           -- String length
    SUBSTRING(name FROM 1 FOR 10) as short_name, -- Extract substring (ANSI syntax)
    TRIM(BOTH ' ' FROM name) as trimmed_name,     -- Remove whitespace
    POSITION('@' IN email) as at_position,        -- Find character position
    
    -- Date functions (ANSI SQL:2016)
    CURRENT_DATE as today,                        -- Current date
    CURRENT_TIMESTAMP as now,                     -- Current timestamp
    EXTRACT(YEAR FROM created_date) as year,      -- Extract date part
    EXTRACT(MONTH FROM created_date) as month,
    EXTRACT(DAY FROM created_date) as day,
    
    -- Aggregate functions (ANSI SQL:1992+)
    COUNT(*) OVER (PARTITION BY category) as category_count,
    SUM(amount) OVER (ORDER BY date) as running_total,
    AVG(rating) OVER (PARTITION BY product_id) as avg_rating,
    
    -- Conditional expressions (ANSI SQL:1992)
    CASE 
        WHEN amount > 1000 THEN 'High'
        WHEN amount > 500 THEN 'Medium'
        ELSE 'Low'
    END as amount_category,
    
    -- NULL handling (ANSI SQL:1992)
    COALESCE(nickname, first_name, 'Unknown') as display_name,
    NULLIF(division, '') as clean_division
    
FROM customers;

ANSI Standard Joins

-- ANSI SQL:1992 Join Syntax (preferred over old-style comma joins)
SELECT 
    c.customer_name,
    o.order_date,
    oi.product_name,
    oi.quantity * oi.unit_price as line_total
FROM customers c
    INNER JOIN orders o ON c.customer_id = o.customer_id
    LEFT OUTER JOIN order_items oi ON o.order_id = oi.order_id
    RIGHT OUTER JOIN products p ON oi.product_id = p.product_id
    FULL OUTER JOIN categories cat ON p.category_id = cat.category_id
WHERE o.order_date >= DATE '2024-01-01'
ORDER BY c.customer_name, o.order_date;

-- Cross join (ANSI SQL:1992)
SELECT d.date, p.product_name
FROM date_dimension d
CROSS JOIN products p
WHERE d.date BETWEEN '2024-01-01' AND '2024-12-31';

ANSI Window Functions

-- ANSI SQL:2003 Window Functions
SELECT 
    employee_id,
    department,
    salary,
    
    -- Ranking functions (ANSI SQL:2003)
    RANK() OVER (PARTITION BY department ORDER BY salary DESC) as salary_rank,
    DENSE_RANK() OVER (ORDER BY salary DESC) as overall_dense_rank,
    ROW_NUMBER() OVER (PARTITION BY department ORDER BY hire_date) as dept_seniority,
    
    -- Statistical functions
    PERCENT_RANK() OVER (ORDER BY salary) as salary_percentile,
    CUME_DIST() OVER (ORDER BY salary) as cumulative_distribution,
    
    -- Offset functions (ANSI SQL:2003)
    LAG(salary, 1) OVER (ORDER BY hire_date) as previous_salary,
    LEAD(salary, 1) OVER (ORDER BY hire_date) as next_salary,
    FIRST_VALUE(salary) OVER (PARTITION BY department ORDER BY hire_date) as first_dept_salary,
    LAST_VALUE(salary) OVER (
        PARTITION BY department 
        ORDER BY hire_date 
        ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
    ) as last_dept_salary,
    
    -- Frame specification (ANSI SQL:2003)
    SUM(salary) OVER (
        ORDER BY hire_date 
        ROWS BETWEEN 2 PRECEDING AND 2 FOLLOWING
    ) as rolling_5_salary_sum
    
FROM employees;

ANSI Common Table Expressions (CTEs)

-- ANSI SQL:1999 WITH clause (Common Table Expressions)
WITH monthly_sales AS (
    SELECT 
        EXTRACT(YEAR FROM order_date) as year,
        EXTRACT(MONTH FROM order_date) as month,
        SUM(total_amount) as monthly_total,
        COUNT(*) as order_count
    FROM orders
    WHERE order_date >= DATE '2023-01-01'
    GROUP BY EXTRACT(YEAR FROM order_date), EXTRACT(MONTH FROM order_date)
),
sales_with_growth AS (
    SELECT 
        year,
        month,
        monthly_total,
        order_count,
        LAG(monthly_total) OVER (ORDER BY year, month) as previous_month_total,
        monthly_total - LAG(monthly_total) OVER (ORDER BY year, month) as growth_amount
    FROM monthly_sales
)
SELECT 
    year,
    month,
    monthly_total,
    order_count,
    growth_amount,
    CASE 
        WHEN previous_month_total > 0 THEN 
            ROUND((growth_amount / previous_month_total) * 100, 2)
        ELSE NULL
    END as growth_percentage
FROM sales_with_growth
ORDER BY year, month;

-- Recursive CTE (ANSI SQL:1999)
WITH RECURSIVE employee_hierarchy AS (
    -- Anchor member
    SELECT employee_id, name, manager_id, 0 as level
    FROM employees
    WHERE manager_id IS NULL
    
    UNION ALL
    
    -- Recursive member
    SELECT e.employee_id, e.name, e.manager_id, eh.level + 1
    FROM employees e
    INNER JOIN employee_hierarchy eh ON e.manager_id = eh.employee_id
)
SELECT employee_id, name, level,
       REPEAT('  ', level) || name as indented_name
FROM employee_hierarchy
ORDER BY level, name;
