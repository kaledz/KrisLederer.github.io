ðŸ›¡ï¸ NULL Handling Functions

â–¼

Functions for handling NULL values, providing defaults, and checking for NULLs. These are essential for data quality and preventing errors in calculations.

COALESCE - Return First Non-NULL Value

ðŸ”µ SQL Syntax
-- Return first non-null value from list
SELECT 
    COALESCE(column1, column2, 'default') AS result,
    COALESCE(price, 0) AS clean_price,
    COALESCE(name, 'Unknown') AS customer_name
FROM table_name;

-- Multiple column coalesce for data cleaning
SELECT 
    customer_id,
    COALESCE(email_primary, email_secondary, email_backup) AS contact_email,
    COALESCE(phone_mobile, phone_home, phone_work) AS contact_phone
FROM customers;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Using coalesce function
df.select(
    F.coalesce(F.col("column1"), F.col("column2"), F.lit("default")).alias("result"),
    F.coalesce(F.col("price"), F.lit(0)).alias("clean_price"),
    F.coalesce(F.col("name"), F.lit("Unknown")).alias("customer_name")
)

# Multiple column coalesce
df.select(
    "customer_id",
    F.coalesce("email_primary", "email_secondary", "email_backup").alias("contact_email"),
    F.coalesce("phone_mobile", "phone_home", "phone_work").alias("contact_phone")
)

ISNULL / ISNOTNULL - Check for NULL Values

ðŸ”µ SQL Syntax
-- Check for NULL values
SELECT *
FROM customers
WHERE email IS NULL;

SELECT *
FROM customers
WHERE email IS NOT NULL;

-- Count NULL values
SELECT 
    COUNT(*) as total_rows,
    COUNT(email) as non_null_emails,
    COUNT(*) - COUNT(email) as null_emails
FROM customers;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Check for NULL values
df.filter(F.col("email").isNull())
df.filter(F.isnull("email"))

df.filter(F.col("email").isNotNull())
df.filter(F.isnan("email"))  # For NaN values

# Count NULL values
df.select(
    F.count("*").alias("total_rows"),
    F.count("email").alias("non_null_emails"),
    (F.count("*") - F.count("email")).alias("null_emails")
)

NULLIF - Return NULL if Values Match

ðŸ”µ SQL Syntax
-- Return NULL if values match (useful for cleaning)
SELECT 
    NULLIF(status, '') AS clean_status,  -- Convert empty strings to NULL
    NULLIF(age, 0) AS valid_age,         -- Convert 0 to NULL
    NULLIF(score, -1) AS actual_score    -- Convert placeholder -1 to NULL
FROM users;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Using when/otherwise for NULLIF behavior
df.select(
    F.when(F.col("status") == "", None).otherwise(F.col("status")).alias("clean_status"),
    F.when(F.col("age") == 0, None).otherwise(F.col("age")).alias("valid_age"),
    F.when(F.col("score") == -1, None).otherwise(F.col("score")).alias("actual_score")
)
  • COALESCE: Essential for providing default values and combining multiple nullable columns
  • NULL checks: Critical for data quality validation and filtering
  • NULLIF: Useful for converting placeholder values (empty strings, zeros) to proper NULLs
  • Performance tip: NULL checks are generally fast, but consider indexing on nullable columns

Functions for implementing conditional logic in SQL queries. These allow you to create dynamic columns and implement business rules directly in your queries.

CASE WHEN - Complex Conditional Logic

ðŸ”µ SQL Syntax
-- Basic CASE WHEN
SELECT 
    customer_id,
    CASE 
        WHEN age < 18 THEN 'Minor'
        WHEN age < 65 THEN 'Adult'
        ELSE 'Senior'
    END AS age_group,
    
    -- Complex business logic
    CASE 
        WHEN total_purchases > 10000 AND years_active > 5 THEN 'VIP'
        WHEN total_purchases > 5000 THEN 'Premium'
        WHEN total_purchases > 1000 THEN 'Standard'
        ELSE 'Basic'
    END AS customer_tier
FROM customers;

-- CASE with aggregations
SELECT 
    department,
    COUNT(CASE WHEN status = 'active' THEN 1 END) as active_count,
    COUNT(CASE WHEN status = 'inactive' THEN 1 END) as inactive_count
FROM employees
GROUP BY department;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Basic CASE WHEN using when/otherwise
df.select(
    "customer_id",
    F.when(F.col("age") < 18, "Minor")
     .when(F.col("age") < 65, "Adult")
     .otherwise("Senior").alias("age_group"),
    
    # Complex business logic
    F.when((F.col("total_purchases") > 10000) & (F.col("years_active") > 5), "VIP")
     .when(F.col("total_purchases") > 5000, "Premium")
     .when(F.col("total_purchases") > 1000, "Standard")
     .otherwise("Basic").alias("customer_tier")
)

# CASE with aggregations
df.groupBy("department").agg(
    F.count(F.when(F.col("status") == "active", 1)).alias("active_count"),
    F.count(F.when(F.col("status") == "inactive", 1)).alias("inactive_count")
)

IF Function - Simple Conditional

ðŸ”µ SQL Syntax
-- Simple IF statements (Spark SQL)
SELECT 
    customer_id,
    IF(is_premium, 'Premium Customer', 'Regular Customer') AS customer_type,
    IF(age >= 18, age, NULL) AS adult_age,
    IF(email LIKE '%@gmail.com', 'Gmail', 'Other') AS email_provider
FROM customers;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Simple conditional using when/otherwise
df.select(
    "customer_id",
    F.when(F.col("is_premium"), "Premium Customer")
     .otherwise("Regular Customer").alias("customer_type"),
    F.when(F.col("age") >= 18, F.col("age")).otherwise(None).alias("adult_age"),
    F.when(F.col("email").like("%@gmail.com"), "Gmail")
     .otherwise("Other").alias("email_provider")
)
  • CASE WHEN: Most flexible conditional logic, supports multiple conditions and complex expressions
  • IF function: Simpler syntax for binary conditions (true/false scenarios)
  • Performance: Both are optimized by Spark's Catalyst optimizer
  • Readability: Use CASE WHEN for complex logic, IF for simple binary conditions

String Manipulation Functions

ðŸ”µ SQL Syntax
-- Basic string functions
SELECT 
    UPPER(name) AS name_upper,
    LOWER(email) AS email_lower,
    TRIM(description) AS description_clean,
    LENGTH(phone) AS phone_length,
    
    -- String concatenation
    CONCAT(first_name, ' ', last_name) AS full_name,

# Basic string functions
df.select(
    F.upper("name").alias("name_upper"),
    F.lower("email").alias("email_lower"),
    F.trim("description").alias("description_clean"),
    F.length("phone").alias("phone_length"),
    
    # String concatenation
    F.concat("first_name", F.lit(" "), "last_name").alias("full_name"),

)

Advanced String Operations

ðŸ”µ SQL Syntax
-- String splitting and position finding
SELECT 
    SPLIT(full_name, ' ')[0] AS first_name,
    SPLIT(tags, ',') AS tag_array,
    POSITION('@' IN email) AS at_position,
    
    -- String padding and formatting
    LPAD(id, 6, '0') AS padded_id,
    RPAD(code, 10, '-') AS formatted_code,
    
    -- String validation
    CASE 
        WHEN email RLIKE '^[A-Za-z0-9+_.-]+@(.+)$' THEN 'Valid'
        ELSE 'Invalid'
    END AS email_status
FROM users;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# String splitting and position finding
df.select(
    F.split("full_name", " ").getItem(0).alias("first_name"),
    F.split("tags", ",").alias("tag_array"),
    F.instr("email", "@").alias("at_position"),
    
    # String padding and formatting
    F.lpad("id", 6, "0").alias("padded_id"),
    F.rpad("code", 10, "-").alias("formatted_code"),
    
    # String validation
    F.when(F.col("email").rlike(r"^[A-Za-z0-9+_.-]+@(.+)$"), "Valid")
     .otherwise("Invalid").alias("email_status")
)
  • CONCAT vs CONCAT_WS: Use CONCAT_WS for separator-based concatenation to avoid NULL issues
  • TRIM variants: LTRIM, RTRIM for left/right trimming, TRIM for both sides
  • Regular expressions: Powerful for pattern matching but can be performance-intensive
  • String arrays: SPLIT creates arrays that can be processed with array functions

ðŸ“Š Aggregate Functions

â–¼

Functions for aggregating data across rows, essential for analytics and reporting queries.

Basic Aggregations

ðŸ”µ SQL Syntax
-- Standard aggregations
SELECT 
    department,
    COUNT(*) AS total_employees,
    COUNT(DISTINCT employee_id) AS unique_employees,
    SUM(salary) AS total_salary,
    AVG(salary) AS average_salary,
    MIN(hire_date) AS earliest_hire,
    MAX(hire_date) AS latest_hire,
    
    -- Statistical aggregations
    STDDEV(salary) AS salary_std_dev,
    VARIANCE(salary) AS salary_variance,
    
    -- Conditional aggregations
    SUM(CASE WHEN status = 'active' THEN salary ELSE 0 END) AS active_salary_total,
    COUNT(CASE WHEN performance_rating > 4 THEN 1 END) AS high_performers
FROM employees
GROUP BY department;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Standard aggregations
df.groupBy("department").agg(
    F.count("*").alias("total_employees"),
    F.countDistinct("employee_id").alias("unique_employees"),
    F.sum("salary").alias("total_salary"),
    F.avg("salary").alias("average_salary"),
    F.min("hire_date").alias("earliest_hire"),
    F.max("hire_date").alias("latest_hire"),
    
    # Statistical aggregations
    F.stddev("salary").alias("salary_std_dev"),
    F.variance("salary").alias("salary_variance"),
    
    # Conditional aggregations
    F.sum(F.when(F.col("status") == "active", F.col("salary")).otherwise(0)).alias("active_salary_total"),
    F.count(F.when(F.col("performance_rating") > 4, 1)).alias("high_performers")
)

Advanced Aggregations

ðŸ”µ SQL Syntax
-- Percentiles and quantiles
SELECT 
    department,
    PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY salary) AS median_salary,
    PERCENTILE_CONT(0.25) WITHIN GROUP (ORDER BY salary) AS q1_salary,
    PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY salary) AS q3_salary,
    
    -- String aggregations
    COLLECT_LIST(employee_name) AS all_employees,
    COLLECT_SET(skill) AS unique_skills,
    
    -- First/Last values
    FIRST(salary ORDER BY hire_date) AS first_hire_salary,
    LAST(salary ORDER BY hire_date) AS last_hire_salary
FROM employees
GROUP BY department;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F

# Percentiles and quantiles
df.groupBy("department").agg(
    F.expr("percentile_cont(0.5) WITHIN GROUP (ORDER BY salary)").alias("median_salary"),
    F.expr("percentile_cont(0.25) WITHIN GROUP (ORDER BY salary)").alias("q1_salary"),
    F.expr("percentile_cont(0.75) WITHIN GROUP (ORDER BY salary)").alias("q3_salary"),
    
    # Alternative percentile syntax
    F.percentile_approx("salary", 0.5).alias("approx_median"),
    F.percentile_approx("salary", [0.25, 0.5, 0.75]).alias("quartiles"),
    
    # String aggregations
    F.collect_list("employee_name").alias("all_employees"),
    F.collect_set("skill").alias("unique_skills"),
    
    # First/Last values (need window function for ordering)
    F.first("salary").alias("first_salary"),
    F.last("salary").alias("last_salary")
)
  • COUNT vs COUNT DISTINCT: COUNT(*) includes NULLs, COUNT(column) excludes NULLs
  • PERCENTILE_CONT vs PERCENTILE_APPROX: CONT is exact but slower, APPROX is faster for large datasets
  • COLLECT functions: Return arrays, be careful with memory usage on large groups
  • Statistical functions: Useful for data quality checks and outlier detection

Ranking Functions

ðŸ”µ SQL Syntax
-- Ranking functions
SELECT 
    employee_id,
    department,
    salary,
    ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS salary_rank,
    RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS salary_dense_rank,
    DENSE_RANK() OVER (PARTITION BY department ORDER BY salary DESC) AS salary_rank_dense,
    NTILE(4) OVER (PARTITION BY department ORDER BY salary DESC) AS salary_quartile,
    
    -- Percentage ranks
    PERCENT_RANK() OVER (PARTITION BY department ORDER BY salary) AS percentile_rank,
    CUME_DIST() OVER (PARTITION BY department ORDER BY salary) AS cumulative_dist
FROM employees;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F
from pyspark.sql.window import Window

# Define window specification
window_spec = Window.partitionBy("department").orderBy(F.desc("salary"))

# Ranking functions
df.select(
    "employee_id", "department", "salary",
    F.row_number().over(window_spec).alias("salary_rank"),
    F.rank().over(window_spec).alias("salary_dense_rank"),
    F.dense_rank().over(window_spec).alias("salary_rank_dense"),
    F.ntile(4).over(window_spec).alias("salary_quartile"),

    F.percent_rank().over(Window.partitionBy("department").orderBy("salary")).alias("percentile_rank"),
    F.cume_dist().over(Window.partitionBy("department").orderBy("salary")).alias("cumulative_dist")
)

Lag/Lead and Offset Functions

ðŸ”µ SQL Syntax
-- Lag/Lead functions for time series analysis
SELECT 
    date,
    revenue,
    LAG(revenue, 1) OVER (ORDER BY date) AS previous_day_revenue,
    LEAD(revenue, 1) OVER (ORDER BY date) AS next_day_revenue,
    
    -- Calculate changes
    revenue - LAG(revenue, 1) OVER (ORDER BY date) AS daily_change,
    (revenue - LAG(revenue, 1) OVER (ORDER BY date)) / LAG(revenue, 1) OVER (ORDER BY date) * 100 AS pct_change,
    
    -- First/Last values in window
    FIRST_VALUE(revenue) OVER (ORDER BY date ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS first_revenue,
    LAST_VALUE(revenue) OVER (ORDER BY date ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS last_revenue
FROM daily_sales;
ðŸ”´ PySpark Implementation
from pyspark.sql import functions as F
from pyspark.sql.window import Window

# Define window specifications
time_window = Window.orderBy("date")
unbounded_window = Window.orderBy("date").rowsBetween(Window.unboundedPreceding, Window.currentRow)
full_window = Window.orderBy("date").rowsBetween(Window.currentRow, Window.unboundedFollowing)

# Lag/Lead functions
df.select(
    "date", "revenue",
    F.lag("revenue", 1).over(time_window).alias("previous_day_revenue"),
    F.lead("revenue", 1).over(time_window).alias("next_day_revenue"),
    
    # Calculate changes
    (F.col("revenue") - F.lag("revenue", 1).over(time_window)).alias("daily_change"),
    ((F.col("revenue") - F.lag("revenue", 1).over(time_window)) / F.lag("revenue", 1).over(time_window) * 100).alias("pct_change"),
    
    # First/Last values in window
    F.first("revenue").over(unbounded_window).alias("first_revenue"),
    F.last("revenue").over(full_window).alias("last_revenue")
)

Moving Averages and Running Totals

ðŸ”µ SQL Syntax
-- Moving averages and running calculations
SELECT 
    date,
    sales,
    
    -- Running totals
    SUM(sales) OVER (ORDER BY date ROWS UNBOUNDED PRECEDING) AS running_total,
    
    -- Moving averages
    AVG(sales) OVER (ORDER BY date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS seven_day_avg,
    AVG(sales) OVER (ORDER BY date ROWS BETWEEN 29 PRECEDING AND CURRENT ROW) AS thirty_day_avg,
    
    -- Rolling statistics
    MIN(sales) OVER (ORDER BY date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS seven_day_min,
    MAX(sales) OVER (ORDER BY date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS seven_day_max,
    STDDEV(sales) OVER (ORDER BY date ROWS BETWEEN 6 PRECEDING AND CURRENT ROW) AS seven_day_stddev
FROM daily_sales;

from pyspark.sql.window import Window

seven_day_window = Window.orderBy("date").rowsBetween(-6, Window.currentRow)
thirty_day_window = Window.orderBy("date").rowsBetween(-29, Window.currentRow)

ðŸ’¡ Pro Tip: Window functions are incredibly powerful for time series analysis. Use them instead of self-joins for better performance when calculating running totals, moving averages, and period-over-period comparisons.
  • ROW_NUMBER vs RANK: ROW_NUMBER gives unique ranks, RANK can have ties
  • ROWS vs RANGE: ROWS counts physical rows, RANGE considers logical ranges based on values
  • Performance: Window functions are optimized but can be memory-intensive for large windows
  • Order matters: Always specify ORDER BY in window functions for consistent results
