Understanding how to properly exit notebooks with different status codes and messages. Proper exit handling is crucial for workflow orchestration and debugging in production environments.

Simple Conditional Exits

# Basic conditional exit based on data validation
def validate_input_data(df):
    """Validate input data and exit if invalid"""
    
    # Check if DataFrame is empty
    if df.count() == 0:
        error_msg = "ERROR: Input dataset is empty. Cannot proceed with processing."
        print(f"❌ {error_msg}")
        dbutils.notebook.exit(error_msg)
    
    # Check for required columns
    required_columns = ["id", "timestamp", "value"]
    missing_columns = set(required_columns) - set(df.columns)
    
    if missing_columns:
        error_msg = f"ERROR: Missing required columns: {missing_columns}"
        print(f"❌ {error_msg}")
        dbutils.notebook.exit(error_msg)
    
    # Check data quality thresholds
    null_percentage = df.filter(col("value").isNull()).count() / df.count()
    
    if null_percentage > 0.5:  # More than 50% nulls
        error_msg = f"ERROR: Data quality too low. {null_percentage:.2%} null values in 'value' column"
        print(f"❌ {error_msg}")
        dbutils.notebook.exit(error_msg)
    
    print("✅ Input data validation passed")
    return True

try:

except Exception as e:
    error_msg = f"UNEXPECTED_ERROR: {str(e)}"

    dbutils.notebook.exit(error_msg)

class NotebookExitCodes:
    SUCCESS = "SUCCESS"

def exit_with_status(status_code, message, additional_info=None):
    """Standardized exit function with logging"""
    import json
    from datetime import datetime
    
    exit_info = {
        "status_code": status_code,
        "message": message,
        "timestamp": datetime.now().isoformat(),
        "notebook_path": dbutils.notebook.entry_point.getDbutils().notebook().getContext().notebookPath().get(),
        "additional_info": additional_info or {}
    }
    
    # Log to console for debugging
    print(f"📋 Notebook Exit Status: {status_code}")
    print(f"📝 Message: {message}")
    
    if additional_info:
        print(f"ℹ️  Additional Info: {json.dumps(additional_info, indent=2)}")
    
    # Exit with JSON formatted status for parsing
    dbutils.notebook.exit(json.dumps(exit_info))

            return "development"

def environment_aware_exit(error_condition, message):

    env = get_environment()

        # In production, always exit on errors

        dbutils.notebook.exit(f"PRODUCTION_ERROR: {message}")
        
    elif env == "staging":
        # In staging, log warning but continue with reduced dataset
        print(f"⚠️  STAGING WARNING: {message}")
        print("📝 Continuing with sample data for testing...")
        return "continue_with_sample"
        
    else:  # development
        # In development, just log and continue
        print(f"🔍 DEV NOTE: {message}")
        print("📝 Continuing for development testing...")
        return "continue"

Basic Retry Decorator

import time
import random
from functools import wraps
from typing import Union, List, Type

def retry_with_backoff(
    max_retries: int = 3,
    base_delay: float = 1.0,
    exponential_base: float = 2.0,
    jitter: bool = True,
    retry_on_exceptions: Union[Type[Exception], List[Type[Exception]]] = Exception
):
    """
    Decorator that retries a function with exponential backoff
    
    Args:
        max_retries: Maximum number of retry attempts
        base_delay: Base delay in seconds before first retry
        exponential_base: Multiplier for exponential backoff
        jitter: Add randomness to prevent thundering herd
        retry_on_exceptions: Exception types to retry on
    """
    
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            
            # Convert single exception type to list
            if isinstance(retry_on_exceptions, type):
                exceptions_to_retry = [retry_on_exceptions]
            else:
                exceptions_to_retry = retry_on_exceptions
            
            for attempt in range(max_retries + 1):
                try:
                    result = func(*args, **kwargs)
                    
                    # Log success if we had previous failures
                    if attempt > 0:
                        print(f"✅ Function {func.__name__} succeeded on attempt {attempt + 1}")
                    
                    return result
                    
                except Exception as e:
                    last_exception = e
                    
                    # Check if this exception type should trigger retry
                    should_retry = any(isinstance(e, exc_type) for exc_type in exceptions_to_retry)
                    
                    if not should_retry or attempt == max_retries:
                        print(f"❌ Function {func.__name__} failed permanently: {str(e)}")
                        raise e
                    
                    # Calculate delay with exponential backoff and jitter
                    delay = base_delay * (exponential_base ** attempt)
                    
                    if jitter:
                        # Add random jitter (±25% of delay)
                        jitter_range = delay * 0.25
                        delay += random.uniform(-jitter_range, jitter_range)
                    
                    print(f"⚠️  Function {func.__name__} failed (attempt {attempt + 1}): {str(e)}")
                    print(f"🔄 Retrying in {delay:.1f} seconds...")
                    
                    time.sleep(delay)
            
            # This should never be reached, but just in case
            raise last_exception
        
        return wrapper
    return decorator

Notebook-Level Retry with dbutils

def run_notebook_with_retry(
    notebook_path: str,
    timeout_seconds: int = 3600,
    arguments: dict = None,
    max_retries: int = 3,
    retry_delay: int = 300  # 5 minutes between retries
):
    """
    Run a notebook with retry logic and comprehensive error handling
    
    Args:
        notebook_path: Path to the notebook to run
        timeout_seconds: Timeout for notebook execution
        arguments: Parameters to pass to the notebook
        max_retries: Maximum number of retry attempts
        retry_delay: Delay between retries in seconds
    """
    
    arguments = arguments or {}
    
    for attempt in range(max_retries + 1):
        try:
            print(f"🚀 Running notebook {notebook_path} (attempt {attempt + 1})")
            print(f"📋 Arguments: {arguments}")
            
            # Run the notebook
            result = dbutils.notebook.run(
                notebook_path,
                timeout_seconds,
                arguments
            )
            
            # Parse result to check for success
            if result.startswith("SUCCESS"):
                print(f"✅ Notebook completed successfully: {result}")
                return result
            elif result.startswith("ERROR") or result.startswith("FAILED"):
                print(f"❌ Notebook failed with error: {result}")
                
                # Check if this is a retryable error
                if is_retryable_error(result) and attempt < max_retries:
                    print(f"🔄 Error appears retryable, will retry in {retry_delay} seconds")
                    time.sleep(retry_delay)
                    continue
                else:
                    # Non-retryable error or max retries reached
                    error_msg = f"Notebook {notebook_path} failed permanently: {result}"
                    print(f"❌ {error_msg}")
                    dbutils.notebook.exit(error_msg)
            else:
                # Unexpected result format
                print(f"⚠️  Unexpected notebook result: {result}")
                return result
                
        except Exception as e:
            error_msg = f"Failed to run notebook {notebook_path}: {str(e)}"
            print(f"❌ {error_msg}")
            
            if attempt < max_retries:
                print(f"🔄 Retrying in {retry_delay} seconds...")
                time.sleep(retry_delay)
            else:
                dbutils.notebook.exit(f"RETRY_EXHAUSTED: {error_msg}")
    
    # Should not reach here
    dbutils.notebook.exit("UNEXPECTED_END: Retry loop ended unexpectedly")

def is_retryable_error(error_message: str) -> bool:
    """Determine if an error message indicates a retryable condition"""
    
    retryable_indicators = [
        "TIMEOUT",
        "CONNECTION_ERROR", 
        "TEMPORARY_UNAVAILABLE",
        "RESOURCE_CONTENTION",
        "CLUSTER_OVERLOADED",
        "NETWORK_ERROR"
    ]
    
    error_upper = error_message.upper()
    
    for indicator in retryable_indicators:
        if indicator in error_upper:
            return True
    
    return False

Circuit Breaker Pattern

from datetime import datetime, timedelta
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Circuit is open, failing fast
    HALF_OPEN = "half_open"  # Testing if service is back

class CircuitBreaker:
    """
    Circuit breaker pattern implementation for notebook operations
    Prevents cascading failures by temporarily disabling failing operations
    """
    
    def __init__(self, 
                 failure_threshold: int = 5,
                 recovery_timeout: int = 300,  # 5 minutes
                 expected_exception: type = Exception):
        
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.expected_exception = expected_exception
        
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
        self.last_success_time = datetime.now()
    
    def call(self, func, *args, **kwargs):
        """Execute function with circuit breaker protection"""
        
        # Check if we should attempt to reset circuit
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
                print("🔄 Circuit breaker moving to HALF_OPEN state")
            else:
                error_msg = f"Circuit breaker is OPEN for {func.__name__}. Fast failing."
                print(f"⚡ {error_msg}")
                raise Exception(error_msg)
        
        try:
            # Execute the function
            result = func(*args, **kwargs)
            self._on_success()
            return result
            
        except self.expected_exception as e:
            self._on_failure()
            print(f"❌ Circuit breaker recorded failure for {func.__name__}: {str(e)}")
            raise e
    
    def _should_attempt_reset(self):
        """Check if enough time has passed to attempt reset"""
        if self.last_failure_time is None:
            return True
        
        time_since_failure = datetime.now() - self.last_failure_time
        return time_since_failure.total_seconds() >= self.recovery_timeout
    
    def _on_success(self):
        """Handle successful execution"""
        if self.state == CircuitState.HALF_OPEN:
            print("✅ Circuit breaker reset to CLOSED state after successful execution")
        
        self.failure_count = 0
        self.state = CircuitState.CLOSED
        self.last_success_time = datetime.now()
    
    def _on_failure(self):
        """Handle failed execution"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
            print(f"⚡ Circuit breaker OPENED after {self.failure_count} failures")
    
    def get_status(self):
        """Get current circuit breaker status"""
        return {
            "state": self.state.value,
            "failure_count": self.failure_count,
            "last_failure_time": self.last_failure_time.isoformat() if self.last_failure_time else None,
            "last_success_time": self.last_success_time.isoformat()
        }

Conditional Workflow Branching

def execute_conditional_workflow(processing_date: str, data_volume_threshold: int = 1000000):
    """
    Execute different workflow branches based on data characteristics
    
    Args:
        processing_date: Date to process (YYYY-MM-DD format)
        data_volume_threshold: Threshold for determining processing strategy
    """
    
    try:
        # Step 1: Analyze input data characteristics
        data_stats = analyze_input_data(processing_date)
        
        if data_stats is None:
            dbutils.notebook.exit("NO_DATA: No data found for processing date")
        
        volume = data_stats["record_count"]
        complexity = data_stats["complexity_score"]  # Custom metric
        
        print(f"📊 Data Analysis Results:")
        print(f"   Volume: {volume:,} records")
        print(f"   Complexity Score: {complexity}")
        
        # Step 2: Choose processing strategy based on data characteristics
        if volume < 1000:
            # Small dataset - single node processing
            strategy = "single_node"
            notebook_path = "/workflows/single_node_processing"
            timeout = 900  # 15 minutes
            
        elif volume < data_volume_threshold:
            # Medium dataset - standard distributed processing
            strategy = "standard_distributed"
            notebook_path = "/workflows/standard_distributed_processing"
            timeout = 3600  # 1 hour
            
        else:
            # Large dataset - optimized processing with multiple strategies
            if complexity > 0.8:
                strategy = "large_complex"
                notebook_path = "/workflows/large_complex_processing"
                timeout = 7200  # 2 hours
            else:
                strategy = "large_simple"
                notebook_path = "/workflows/large_simple_processing"  
                timeout = 5400  # 1.5 hours
        
        print(f"🎯 Selected processing strategy: {strategy}")
        
        # Step 3: Execute chosen strategy with appropriate parameters
        processing_args = {
            "processing_date": processing_date,
            "volume": volume,
            "complexity": complexity,
            "strategy": strategy
        }
        
        # Add strategy-specific parameters
        if strategy == "single_node":
            processing_args["executor_memory"] = "2g"
            processing_args["executor_cores"] = "1"
        elif strategy.startswith("large"):
            processing_args["executor_memory"] = "8g"
            processing_args["executor_cores"] = "4"
            processing_args["dynamic_allocation"] = "true"
        
        # Execute processing notebook
        result = dbutils.notebook.run(
            notebook_path,
            timeout,
            processing_args
        )
        
        # Step 4: Validate results and determine next steps
        if result.startswith("SUCCESS"):
            print(f"✅ Processing completed successfully with strategy: {strategy}")
            
            # Execute post-processing based on strategy
            post_process_result = execute_post_processing(strategy, processing_date)
            
            if post_process_result.startswith("SUCCESS"):
                dbutils.notebook.exit(f"SUCCESS: Workflow completed with {strategy} strategy")
            else:
                dbutils.notebook.exit(f"POST_PROCESSING_FAILED: {post_process_result}")
                
        else:
            # Processing failed - try fallback strategy if available
            fallback_result = attempt_fallback_processing(strategy, processing_date, result)
            
            if fallback_result.startswith("SUCCESS"):
                dbutils.notebook.exit(f"SUCCESS: Completed with fallback processing")
            else:
                dbutils.notebook.exit(f"FAILED: Both primary and fallback processing failed")
        
    except Exception as e:
        error_msg = f"Workflow orchestration failed: {str(e)}"
        print(f"❌ {error_msg}")
        dbutils.notebook.exit(f"ORCHESTRATION_ERROR: {error_msg}")

def analyze_input_data(processing_date: str) -> dict:
    """Analyze input data to determine processing strategy"""
    
    input_path = f"/mnt/raw/{processing_date}/"
    
    try:
        df = spark.read.parquet(input_path)
        
        record_count = df.count()
        
        if record_count == 0:
            return None
        
        # Calculate complexity score based on schema and data characteristics
        column_count = len(df.columns)
        nested_columns = len([col for col in df.dtypes if col[1].startswith('array') or col[1].startswith('struct')])
        
        # Simple complexity scoring (0.0 to 1.0)
        complexity_score = min(1.0, (nested_columns / column_count) + (column_count / 100))
        
        return {
            "record_count": record_count,
            "column_count": column_count,
            "nested_columns": nested_columns,
            "complexity_score": complexity_score
        }
        
    except Exception as e:
        print(f"❌ Failed to analyze input data: {str(e)}")
        return None

def execute_post_processing(strategy: str, processing_date: str) -> str:
    """Execute strategy-specific post-processing steps"""
    
    post_processing_configs = {
        "single_node": {
            "notebook": "/workflows/post_process_simple",
            "timeout": 300
        },
        "standard_distributed": {
            "notebook": "/workflows/post_process_standard", 
            "timeout": 600
        },
        "large_complex": {
            "notebook": "/workflows/post_process_advanced",
            "timeout": 1800
        },
        "large_simple": {
            "notebook": "/workflows/post_process_bulk",
            "timeout": 1200
        }
    }
    
    config = post_processing_configs.get(strategy)
    
    if not config:
        return f"ERROR: No post-processing configuration for strategy {strategy}"
    
    try:
        result = dbutils.notebook.run(
            config["notebook"],
            config["timeout"],
            {"processing_date": processing_date, "strategy": strategy}
        )
        
        return result
        
    except Exception as e:
        return f"ERROR: Post-processing failed: {str(e)}"

def attempt_fallback_processing(primary_strategy: str, processing_date: str, error_message: str) -> str:
    """Attempt fallback processing when primary strategy fails"""
    
    print(f"🔄 Attempting fallback processing for failed strategy: {primary_strategy}")
    print(f"📝 Original error: {error_message}")
    
    # Define fallback strategies
    fallback_map = {
        "large_complex": "standard_distributed",
        "large_simple": "standard_distributed", 
        "standard_distributed": "single_node"
    }
    
    fallback_strategy = fallback_map.get(primary_strategy)
    
    if not fallback_strategy:
        return f"ERROR: No fallback available for strategy {primary_strategy}"
    
    print(f"🎯 Trying fallback strategy: {fallback_strategy}")
    
    try:
        # Use simpler processing approach
        fallback_notebook = f"/workflows/{fallback_strategy}_processing"
        
        result = dbutils.notebook.run(
            fallback_notebook,
            1800,  # Reduced timeout for fallback
            {
                "processing_date": processing_date,
                "fallback_mode": "true",
                "original_strategy": primary_strategy
            }
        )
        
        if result.startswith("SUCCESS"):
            print(f"✅ Fallback processing succeeded with {fallback_strategy}")
        
        return result
        
    except Exception as e:
        return f"ERROR: Fallback processing also failed: {str(e)}"
