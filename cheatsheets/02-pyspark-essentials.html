1. Basic DataFrame Operations

▼

Fundamental operations for creating, reading, writing, and manipulating DataFrames in PySpark. These operations form the foundation of any ELT pipeline and are essential for data ingestion and basic transformations.

Creating and Reading DataFrames

# Creating DataFrame from Python data structures
from pyspark.sql import SparkSession
from pyspark.sql.functions import *
from pyspark.sql.types import *

# Initialize SparkSession
spark = SparkSession.builder.appName("ELT_Pipeline").getOrCreate()

# Create DataFrame from list of dictionaries
data = [{"name": "Alice", "age": 25}, {"name": "Bob", "age": 30}]
df = spark.createDataFrame(data)

# Reading CSV File with comprehensive options
df_csv = spark.read \
    .option("header", "true") \
    .option("inferSchema", "true") \
    .option("multiLine", "true") \
    .csv("path/to/file.csv")

# Reading JSON File - handles nested structures automatically
df_json = spark.read.json("path/to/file.json")

# Reading Parquet File - most efficient format for analytics
df_parquet = spark.read.parquet("path/to/file.parquet")
  • What it does: Creates DataFrames from various sources and formats
  • When to use: Data ingestion phase of ELT pipelines
  • Performance tip: Parquet is fastest for large datasets; CSV inference can be expensive on big files
  • Edge case: Schema inference works well for small files but can timeout on large datasets

def validate_and_fix_schema(df, expected_schema):
    """Validate DataFrame schema against expected schema"""
    current_schema = df.schema
    
    # Check if schemas match
    if current_schema == expected_schema:
        return df
    
    # Convert types to match expected schema
    for field in expected_schema.fields:
        field_name = field.name
        expected_type = field.dataType
        
        if field_name in df.columns:
            current_type = df.schema[field_name].dataType
            if current_type != expected_type:
                df = df.withColumn(field_name, col(field_name).cast(expected_type))
    
    return df

# Schema evolution - adding new columns
def add_new_columns(df, new_columns):
    """Add new columns to DataFrame with default values"""
    for col_name, col_type, default_value in new_columns:
        df = df.withColumn(col_name, lit(default_value).cast(col_type))
    return df

Advanced Schema Operations

# Schema comparison and validation
def compare_schemas(schema1, schema2):
    """Compare two schemas and return differences"""
    fields1 = {f.name: f.dataType for f in schema1.fields}
    fields2 = {f.name: f.dataType for f in schema2.fields}
    
    common_fields = set(fields1.keys()) & set(fields2.keys())
    only_in_1 = set(fields1.keys()) - set(fields2.keys())
    only_in_2 = set(fields2.keys()) - set(fields1.keys())
    
    type_mismatches = {
        field: (fields1[field], fields2[field])
        for field in common_fields
        if fields1[field] != fields2[field]
    }
    
    return {
        "common_fields": list(common_fields),
        "only_in_schema1": list(only_in_1),
        "only_in_schema2": list(only_in_2),
        "type_mismatches": type_mismatches
    }

# Schema flattening for nested structures
def flatten_schema(schema, prefix=""):
    """Flatten a nested schema into a list of (field_name, data_type) tuples"""
    flattened = []
    
    for field in schema.fields:
        field_name = f"{prefix}.{field.name}" if prefix else field.name
        
        if isinstance(field.dataType, StructType):
            flattened.extend(flatten_schema(field.dataType, field_name))
        elif isinstance(field.dataType, ArrayType):
            if isinstance(field.dataType.elementType, StructType):
                flattened.extend(flatten_schema(field.dataType.elementType, f"{field_name}[*]"))
            else:
                flattened.append((field_name, field.dataType))
        else:
            flattened.append((field_name, field.dataType))
    
    return flattened

# Schema-based data validation
def validate_data_types(df):
    """Validate that data conforms to schema types"""
    validation_results = {}
    
    for field in df.schema.fields:
        field_name = field.name
        expected_type = field.dataType
        
        # Check for null values in non-nullable fields
        if not field.nullable:
            null_count = df.filter(col(field_name).isNull()).count()
            validation_results[f"{field_name}_nulls"] = null_count
        
        # Check for type mismatches (basic check)
        if isinstance(expected_type, (IntegerType, LongType)):
            # Check if numeric values are within reasonable bounds
            invalid_count = df.filter(
                (col(field_name).isNotNull()) & 
                (col(field_name) < -999999999)
            ).count()
            validation_results[f"{field_name}_invalid_range"] = invalid_count
    
    return validation_results

def evolve_schema_safely(df, new_schema):
    """Safely evolve schema while maintaining backward compatibility"""
    current_schema = df.schema
    current_fields = {f.name: f for f in current_schema.fields}
    
    # Add new fields with default values
    for field in new_schema.fields:
        if field.name not in current_fields:
            if isinstance(field.dataType, StringType):
                df = df.withColumn(field.name, lit(""))
            elif isinstance(field.dataType, IntegerType):
                df = df.withColumn(field.name, lit(0))
            elif isinstance(field.dataType, BooleanType):
                df = df.withColumn(field.name, lit(False))

                df = df.withColumn(field.name, array().cast(field.dataType))

                df = df.withColumn(field.name, struct().cast(field.dataType))
    
    return df

def validate_schema(expected_schema):
    """Decorator to validate DataFrame schema"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            result = func(*args, **kwargs)
            if hasattr(result, 'schema'):
                if result.schema != expected_schema:
                    raise ValueError(f"Schema mismatch. Expected: {expected_schema}, Got: {result.schema}")
            return result
        return wrapper
    return decorator

# 7. Schema documentation generator
def generate_schema_documentation(schema, schema_name="Unknown"):
    """Generate documentation for a schema"""
    doc = f"# Schema Documentation: {schema_name}\n\n"
    
    def document_field(field, indent=0):
        spaces = "  " * indent
        nullable = "nullable" if field.nullable else "required"
        doc_line = f"{spaces}- **{field.name}** ({field.dataType}) - {nullable}\n"
        
        if isinstance(field.dataType, StructType):
            doc_line += f"{spaces}  Nested structure:\n"
            for nested_field in field.dataType.fields:
                doc_line += document_field(nested_field, indent + 2)
        elif isinstance(field.dataType, ArrayType):
            doc_line += f"{spaces}  Array of: {field.dataType.elementType}\n"
        elif isinstance(field.dataType, MapType):
            doc_line += f"{spaces}  Map of {field.dataType.keyType} -> {field.dataType.valueType}\n"
        
        return doc_line
    
    for field in schema.fields:
        doc += document_field(field)
    
    return doc

def get_data_size(df, unit='GB'):
    """Get estimated data size in specified unit (B, KB, MB, GB, TB)"""
    # Get row count and column count
    row_count = df.count()
    column_count = len(df.columns)
    
    # Estimate average row size in bytes (rough estimation)
    # Assuming average of 8 bytes per field (typical for mixed data types)
    estimated_bytes_per_row = column_count * 8
    total_bytes = row_count * estimated_bytes_per_row
    
    # Convert to requested unit
    units = {'B': 1, 'KB': 1024, 'MB': 1024**2, 'GB': 1024**3, 'TB': 1024**4}
    if unit.upper() in units:
        size_in_unit = total_bytes / units[unit.upper()]
        return size_in_unit
    else:
        raise ValueError(f"Invalid unit: {unit}. Use B, KB, MB, GB, or TB")
    
    return total_bytes
